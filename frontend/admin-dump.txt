import { Navigate, Outlet } from "react-router-dom";

import useAccount from "../../account/hooks/useAccount";

const ProtectedAdminRoute = () => {
  const {
    currentUser,
    isAuthenticated,
    isLoading,
  } = useAccount();

  if (isLoading) {
    return (
      <div className="admin-auth-loading" role="status">
        Verifying administrator access...
      </div>
    );
  }

  if (!isAuthenticated) {
    return <Navigate to="/login" replace />;
  }

  if (currentUser?.role !== "admin") {
    return <Navigate to="/" replace />;
  }

  return <Outlet />;
};

import { useCallback, useEffect, useState } from "react";

import adminService from "../services/adminService";

const useAdminDashboard = () => {
  const [dashboard, setDashboard] = useState(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState("");

  const fetchDashboard = useCallback(async (signal) => {
    try {
      setError("");

      const data = await adminService.getDashboard({ signal });

      setDashboard(data);
    } catch (error) {
      if (error.code === "ERR_CANCELED") {
        return;
      }

      setError(
        error.response?.data?.message ||
          "Unable to load dashboard data. Please try again."
      );
    } finally {
      if (!signal?.aborted) {
        setIsLoading(false);
      }
    }
  }, []);

  useEffect(() => {
    const controller = new AbortController();

    fetchDashboard(controller.signal);

    return () => controller.abort();
  }, [fetchDashboard]);

  const refreshDashboard = useCallback(() => {
    setIsLoading(true);
    fetchDashboard();
  }, [fetchDashboard]);

  return {
    dashboard,
    isLoading,
    error,
    refreshDashboard,
  };
};

import {
  useCallback,
  useEffect,
  useState,
} from "react";

import adminProductService from "../services/adminProductService";

const INITIAL_FILTERS = {
  search: "",
  category: "",
  status: "all",
  stockStatus: "all",
  sort: "newest",
  page: 1,
  limit: 12,
};

const useAdminProducts = () => {
  const [products, setProducts] = useState([]);
  const [categories, setCategories] = useState([]);
  const [pagination, setPagination] = useState(null);

  const [filters, setFilters] = useState(INITIAL_FILTERS);
  const [searchInput, setSearchInput] = useState("");

  const [isLoading, setIsLoading] = useState(true);
  const [actionLoading, setActionLoading] = useState("");
  const [error, setError] = useState("");

  const fetchProducts = useCallback(async (signal) => {
    try {
      setError("");

      const params = {
        ...filters,
        category: filters.category || undefined,
      };

      const data = await adminProductService.getProducts(
        params,
        { signal }
      );

      setProducts(data.products);
      setCategories(data.categories);
      setPagination(data.pagination);
    } catch (requestError) {
      if (
        requestError.code === "ERR_CANCELED" ||
        signal?.aborted
      ) {
        return;
      }

      setError(
        requestError.response?.data?.message ||
          "Unable to load products. Please try again."
      );
    } finally {
      if (!signal?.aborted) {
        setIsLoading(false);
      }
    }
  }, [filters]);

  useEffect(() => {
    const controller = new AbortController();

    setIsLoading(true);
    fetchProducts(controller.signal);

    return () => controller.abort();
  }, [fetchProducts]);

  useEffect(() => {
    const timeout = setTimeout(() => {
      setFilters((current) => {
        if (current.search === searchInput.trim()) {
          return current;
        }

        return {
          ...current,
          search: searchInput.trim(),
          page: 1,
        };
      });
    }, 350);

    return () => clearTimeout(timeout);
  }, [searchInput]);

  const updateFilter = useCallback((name, value) => {
    setFilters((current) => ({
      ...current,
      [name]: value,
      page: 1,
    }));
  }, []);

  const changePage = useCallback((page) => {
    setFilters((current) => ({
      ...current,
      page,
    }));
  }, []);

  const refreshProducts = useCallback(() => {
    setIsLoading(true);
    fetchProducts();
  }, [fetchProducts]);

  const toggleProductStatus = useCallback(async (product) => {
    const productId = product._id;

    try {
      setActionLoading(productId);
      setError("");

      if (product.isActive) {
        await adminProductService.deactivateProduct(productId);
      } else {
        await adminProductService.activateProduct(productId);
      }

      await fetchProducts();
    } catch (requestError) {
      setError(
        requestError.response?.data?.message ||
          "Unable to update product status. Please try again."
      );
    } finally {
      setActionLoading("");
    }
  }, [fetchProducts]);

  return {
    products,
    categories,
    pagination,
    filters,
    searchInput,
    setSearchInput,
    updateFilter,
    changePage,
    refreshProducts,
    toggleProductStatus,
    isLoading,
    actionLoading,
    error,
  };
};

export default useAdminProducts;

              url: image.url,
              publicId: image.publicId,
              file: null,
              previewUrl: image.url,
              isExisting: true,
            }))
          : [];

        const initialImages = existingImages.length
          ? existingImages
          : product.image
            ? [
                {
                  id: "legacy-image",
                  url: product.image,
                  publicId: null,
                  file: null,
                  previewUrl: product.image,
                  isExisting: true,
                  isLegacy: true,
                },
              ]
            : [];

        setImages(initialImages);
        setMainImageId(initialImages[0]?.id ?? null);
        setSlugTouched(true);
      } catch (requestError) {
        if (
          requestError.code === "ERR_CANCELED" ||
          controller.signal.aborted
        ) {
          return;
        }

        setError(getErrorMessage(requestError));
      } finally {
        if (!controller.signal.aborted) {
          setLoading(false);
        }
      }
    };

    loadProduct();

    return () => controller.abort();
  }, [id, isEditMode]);

  const updateField = (name, value) => {
    setForm((current) => {
      const updated = { ...current, [name]: value };

      if (name === "name" && !slugTouched) {
        updated.slug = slugify(value);
      }

      return updated;
    });
  };

  const handleCategoryChange = (event) => {
    const categoryId = event.target.value;

    const selectedCategory = categories.find(
      (category) => category._id === categoryId
    );

    if (selectedCategory) {
      setForm((current) => ({
        ...current,
        categoryId: selectedCategory._id,
        category: selectedCategory.name,
      }));

      return;
    }

    // Preserve an existing legacy category.
    setForm((current) => ({
      ...current,
      categoryId: "",
    }));
  };

  const releaseImagePreview = (url) => {
    if (!url || !objectUrlsRef.current.has(url)) return;

    URL.revokeObjectURL(url);
    objectUrlsRef.current.delete(url);
  };

  const handleImageSelection = (event) => {
    const selectedFiles = Array.from(event.target.files || []);

    event.target.value = "";

    if (!selectedFiles.length) return;

    setError("");

    const availableSlots = MAX_IMAGES - images.length;

    if (availableSlots <= 0) {
      setError(`Maximum ${MAX_IMAGES} images allowed hain.`);
      return;
    }

    const acceptedFiles = [];
    const validationErrors = [];

    selectedFiles.forEach((file) => {
      if (!ALLOWED_IMAGE_TYPES.includes(file.type)) {
        validationErrors.push(
          `${file.name}: Unsupported image format.`
        );
        return;
      }

      if (file.size > MAX_IMAGE_SIZE) {
        validationErrors.push(
          `${file.name}: Image size 5MB se zyada hai.`
        );
        return;
      }

      acceptedFiles.push(file);
    });

    const filesToAdd = acceptedFiles.slice(0, availableSlots);

    if (acceptedFiles.length > availableSlots) {
      validationErrors.push(
        `Sirf ${availableSlots} additional images add ho sakti hain.`
      );
    }

    if (validationErrors.length) {
      setError(validationErrors.join(" "));
    }

    if (!filesToAdd.length) return;

    const newImages = filesToAdd.map((file, index) => {
      const previewUrl = URL.createObjectURL(file);

      objectUrlsRef.current.add(previewUrl);

      return {
        id: createImageId(file, index),
        url: "",
        publicId: null,
        file,
        previewUrl,
        isExisting: false,
      };
    });

    if (!images.some((image) => !image.isLegacy)) {
      setMainImageId(newImages[0].id);
    }

    setImages((current) => [...current, ...newImages]);
  };

  const handleRemoveImage = (imageId) => {
    const targetImage = images.find(
      (image) => image.id === imageId
    );

    if (!targetImage) return;

    releaseImagePreview(targetImage.previewUrl);

    const updatedImages = images.filter(
      (image) => image.id !== imageId
    );

    setImages(updatedImages);

    if (mainImageId === imageId) {
      setMainImageId(updatedImages[0]?.id ?? null);
    }
  };

  const handleSetMainImage = (imageId) => {
    const targetImage = images.find(
      (image) => image.id === imageId
    );

    if (!targetImage) return;

    if (
      targetImage.isLegacy &&
      images.some((image) => !image.isLegacy)
    ) {
      setError(
        "Cloudinary images mein se main image select karein."
      );
      return;
    }

    setMainImageId(imageId);
  };

  const handleSubmit = async (event) => {
    event.preventDefault();

    setError("");

    const mainImage =
      images.find((image) => image.id === mainImageId) ??
      images[0] ??
      null;

    if (!mainImage) {
      setError("Kam az kam aik product image select karein.");
      return;
    }

    if (mainImage.id !== mainImageId) {
      setMainImageId(mainImage.id);
    }

    const payload = {
      ...form,
      sku: form.sku.trim().toUpperCase(),
      slug: form.slug.trim(),
      name: form.name.trim(),
      category: form.category.trim(),
      price: Number(form.price),
      compareAtPrice:
        form.compareAtPrice.trim() === ""
          ? null
          : Number(form.compareAtPrice),
      badge: form.badge.trim() || null,
      description: form.description.trim(),
      stock: Number(form.stock),
      isNewArrival: Boolean(form.isNewArrival),
      bestSeller: Boolean(form.bestSeller),
      isActive: Boolean(form.isActive),
    };

    const selectedCategory = categories.find(
      (category) =>
        category._id === form.categoryId ||
        category.name === form.category
    );

    if (selectedCategory) {
      payload.categoryId = selectedCategory._id;
      payload.category = selectedCategory.name;
    } else if (form.categoryId) {
      setError(
        "Selected category available nahi hai. Dobara select karein."
      );
      return;
    } else {
      delete payload.categoryId;
    }

    if (
      !Number.isFinite(payload.price) ||
      payload.price < 0 ||
      !Number.isInteger(payload.stock) ||
      payload.stock < 0
    ) {
      setError(
        "Price aur stock valid non-negative numbers hone chahiye."
      );
      return;
    }

    if (
      payload.compareAtPrice !== null &&
      (!Number.isFinite(payload.compareAtPrice) ||
        payload.compareAtPrice <= payload.price)
    ) {
      setError(
        "Compare-at price selling price se zyada honi chahiye."
      );
      return;
    }

    try {
      setSaving(true);

      const newImages = images.filter(
        (image) => image.file
      );

      let uploadedImages = [];

      if (newImages.length) {
        uploadedImages =
          await adminProductService.uploadProductImages(
            newImages.map((image) => image.file)
          );
      }

      let uploadIndex = 0;

      const resolvedImages = images.map((image) => {
        if (!image.file) return image;

        const uploadedImage = uploadedImages[uploadIndex];

        uploadIndex += 1;

        if (!uploadedImage?.url || !uploadedImage?.publicId) {
          throw new Error(
            "Image upload response invalid hai. Dobara try karein."
          );
        }

        return {
          ...image,
          url: uploadedImage.url,
          publicId: uploadedImage.publicId,
        };
      });

      const orderedImages = [
        ...resolvedImages.filter(
          (image) => image.id === mainImageId
        ),
        ...resolvedImages.filter(
          (image) => image.id !== mainImageId
        ),
      ];

      const primaryImage = orderedImages[0];

      payload.image = primaryImage.url;

      const cloudinaryImages = orderedImages
        .filter((image) => image.publicId)
        .map(({ url, publicId }) => ({
          url,
          publicId,
        }));

      payload.images = cloudinaryImages;

      if (isEditMode) {
        delete payload.sku;

        await adminProductService.updateProduct(
          id,
          payload
        );
      } else {
        await adminProductService.createProduct(payload);
      }

      navigate("/admin/products", {
        replace: true,
        state: {
          message: isEditMode
            ? "Product updated successfully."
            : "Product created successfully.",
        },
      });
    } catch (requestError) {
      setError(getErrorMessage(requestError));
    } finally {
      setSaving(false);
    }
  };

  if (loading) {
    return (
      <main className="admin-product-form">
        <div className="admin-product-form__state">
          <LoaderCircle className="is-spinning" size={28} />
          <span>Loading product details...</span>
        </div>
      </main>
    );
  }

  return (
    <main className="admin-product-form">
      <div className="admin-product-form__container">
        <header className="admin-product-form__header">
          <div>
            <Link
              to="/admin/products"
              className="admin-product-form__back"
            >
              <ArrowLeft size={16} />
              Back to products
            </Link>

            <span className="admin-product-form__eyebrow">
              NEXAMART ADMINISTRATION
            </span>

            <h1>
              {isEditMode ? "Edit Product" : "Create Product"}
            </h1>

            <p>
              {isEditMode
                ? "Update your product information and catalog settings."
                : "Add a new product to your NexaMart catalog."}
            </p>
          </div>

          <div className="admin-product-form__security">
            <ShieldCheck size={17} />
            Authorized Administrator
          </div>
        </header>

        {error && (
          <div className="admin-product-form__error" role="alert">
            {error}
          </div>
        )}

        <form
          className="admin-product-form__layout"
          onSubmit={handleSubmit}
        >
          <section className="admin-product-form__panel">
            <div className="admin-product-form__section-heading">
              <h2>Product Information</h2>
              <p>Basic details displayed in your store catalog.</p>
            </div>

            <div className="admin-product-form__grid">
              <label className="admin-product-form__field">
                <span>Product Name *</span>
                <input
                  type="text"
                  value={form.name}
                  onChange={(event) =>
                    updateField("name", event.target.value)
                  }
                  minLength={2}
                  maxLength={160}
                  required
                  placeholder="Enter product name"
                />
              </label>

              <label className="admin-product-form__field">
                <span>SKU *</span>
                <input
                  type="text"
                  value={form.sku}
                  onChange={(event) =>
                    updateField(
                      "sku",
                      event.target.value.toUpperCase()
                    )
                  }
                  minLength={3}
                  maxLength={40}
                  required
                  readOnly={isEditMode}
                  placeholder="e.g. NM-009"
                />
                {isEditMode && (
                  <small>SKU cannot be changed after creation.</small>
                )}
              </label>

              <label className="admin-product-form__field">
                <span>Product Slug *</span>
                <input
                  type="text"
                  value={form.slug}
                  onChange={(event) => {
                    setSlugTouched(true);
                    updateField("slug", slugify(event.target.value));
                  }}
                  minLength={3}
                  maxLength={180}
                  pattern="[a-z0-9]+(?:-[a-z0-9]+)*"
                  required
                  placeholder="product-url-slug"
                />
                <small>Used in the product URL.</small>
              </label>

              <label className="admin-product-form__field">
                <span>Category *</span>
                <select
                  value={
                    form.categoryId ||
                    (form.category ? `legacy:${form.category}` : "")
                  }
                  onChange={handleCategoryChange}
                  required
                  disabled={categoriesLoading || saving}
                >
                  <option value="">
                    {categoriesLoading
                      ? "Loading categories..."
                      : "Select a category"}
                  </option>

                  {categories.map((category) => (
                    <option
                      key={category._id}
                      value={category._id}
                    >
                      {category.name}
                    </option>
                  ))}

                  {form.category &&
                    !categories.some(
                      (category) =>
                        category._id === form.categoryId ||
                        category.name === form.category
                    ) && (
                      <option value={`legacy:${form.category}`}>
                        {form.category} (Legacy)
                      </option>
                    )}
                </select>
              </label>

              <label className="admin-product-form__field">
                <span>Badge</span>
                <input
                  type="text"
                  value={form.badge}
                  onChange={(event) =>
                    updateField("badge", event.target.value)
                  }
                  maxLength={60}
                  placeholder="e.g. Best Value"
                />
              </label>
            </div>

            <label className="admin-product-form__field admin-product-form__description">
              <span>Description *</span>
              <textarea
                value={form.description}
                onChange={(event) =>
                  updateField("description", event.target.value)
                }
                minLength={10}
                maxLength={3000}
                rows={6}
                required
                placeholder="Write a detailed product description..."
              />
              <small>
                {form.description.length}/3000 characters
              </small>
            </label>
          </section>

          <section className="admin-product-form__panel">
            <div className="admin-product-form__section-heading">
              <h2>Product Images</h2>
              <p>
                Computer se images select karein. Main image
                storefront par primary image hogi.

            <div className="admin-product-form__upload">
              <input
                ref={fileInputRef}

                accept="image/jpeg,image/png,image/webp,image/avif"
                multiple
                hidden
                onChange={handleImageSelection}

              <button
                type="button"

                onClick={() => fileInputRef.current?.click()}
                disabled={saving || images.length >= MAX_IMAGES}

                <Upload size={19} />

                  {images.length >= MAX_IMAGES
                    ? "Image limit reached"
                    : "Choose images from computer"}

              </button>

              <p className="admin-product-form__upload-hint">
                JPEG, PNG, WEBP, AVIF Â· Maximum 5MB each Â·
                Maximum 8 images
              </p>
            </div>

            <div className="admin-product-form__image-count">
              <span>Selected images</span>
              <strong>
                {images.length}/{MAX_IMAGES}

            {images.length ? (
              <div className="admin-product-form__image-grid">
                {images.map((image) => {

                  return (
                    <article
                      className={`admin-product-form__image-card ${
                        isMain ? "is-main" : ""
                      }`}
                      key={image.id}

                      <div className="admin-product-form__image-frame">
                        <img

                      </div>

                      <div className="admin-product-form__image-meta">
                        {!isMain && (
                          <button

                              handleSetMainImage(image.id)

                          <small>Existing legacy image</small>
                        )}
                      </div>
                    </article>

              </div>
            ) : (
              <div className="admin-product-form__empty-images">
                <ImageIcon size={34} />
                <span>No product images selected.</span>
                <small>At least one image is required.</small>
              </div>
            )}
          </section>

          <section className="admin-product-form__panel">
            <div className="admin-product-form__section-heading">
              <h2>Pricing & Inventory</h2>
              <p>Configure product pricing and stock availability.</p>
            </div>

            <div className="admin-product-form__grid">
              <label className="admin-product-form__field">
                <span>Selling Price (PKR) *</span>
                <input
                  type="number"
                  min="0"
                  step="0.01"
                  value={form.price}
                  onChange={(event) =>
                    updateField("price", event.target.value)
                  }
                  required
                  placeholder="0"
                />
              </label>

              <label className="admin-product-form__field">
                <span>Compare-at Price (PKR)</span>
                <input
                  type="number"
                  min="0"
                  step="0.01"
                  value={form.compareAtPrice}
                  onChange={(event) =>
                    updateField("compareAtPrice", event.target.value)
                  }
                  placeholder="Optional"
                />
                <small>Must be higher than selling price.</small>
              </label>

              <label className="admin-product-form__field">
                <span>Stock Quantity *</span>
                <input
                  type="number"
                  min="0"
                  step="1"
                  value={form.stock}
                  onChange={(event) =>
                    updateField("stock", event.target.value)
                  }
                  required
                />
              </label>
            </div>
          </section>

          <section className="admin-product-form__panel">
            <div className="admin-product-form__section-heading">
              <h2>Catalog Settings</h2>
              <p>Control product visibility and merchandising.</p>
            </div>

            <div className="admin-product-form__toggles">
              <label>
                <input
                  type="checkbox"
                  checked={form.isNewArrival}
                  onChange={(event) =>
                    updateField("isNewArrival", event.target.checked)
                  }
                />
                <span>
                  <strong>New Arrival</strong>
                  <small>Display this product in new arrivals.</small>
                </span>
              </label>

              <label>
                <input
                  type="checkbox"
                  checked={form.bestSeller}
                  onChange={(event) =>
                    updateField("bestSeller", event.target.checked)
                  }
                />
                <span>
                  <strong>Best Seller</strong>
                  <small>Mark this product as a best seller.</small>
                </span>
              </label>

              <label>
                <input
                  type="checkbox"
                  checked={form.isActive}
                  onChange={(event) =>
                    updateField("isActive", event.target.checked)
                  }
                />
                <span>
                  <strong>Active Product</strong>
                  <small>Make this product available in the store.</small>
                </span>
              </label>
            </div>
          </section>

          <div className="admin-product-form__actions">
            <Link
              to="/admin/products"
              className="admin-product-form__cancel"
            >
              Cancel
            </Link>

            <button
              type="submit"
              className="admin-product-form__submit"
              disabled={saving || categoriesLoading}
            >
              {saving ? (
                <LoaderCircle size={17} className="is-spinning" />
              ) : (
                <Save size={17} />
              )}

              {saving
                ? "Saving..."
                : isEditMode
                  ? "Save Changes"
                  : "Create Product"}
            </button>
          </div>
        </form>
      </div>
    </main>
  );
};

export default AdminProductForm;

const ProtectedAdminRoute = () => {

  } = useAccount();

    return (

  return <Outlet />;

import { useCallback, useEffect, useState } from "react";

import adminService from "../services/adminService";

const useAdminDashboard = () => {
  const [dashboard, setDashboard] = useState(null);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState("");

  const fetchDashboard = useCallback(async (signal) => {
    try {
      setError("");

      const data = await adminService.getDashboard({ signal });

      setDashboard(data);
    } catch (error) {
      if (error.code === "ERR_CANCELED") {
        return;
      }

      setError(
        error.response?.data?.message ||
          "Unable to load dashboard data. Please try again."
      );
    } finally {
      if (!signal?.aborted) {
        setIsLoading(false);
      }
    }
  }, []);

  useEffect(() => {
    const controller = new AbortController();

    fetchDashboard(controller.signal);

    return () => controller.abort();
  }, [fetchDashboard]);

  const refreshDashboard = useCallback(() => {
    setIsLoading(true);
    fetchDashboard();
  }, [fetchDashboard]);

  return {
    dashboard,
    isLoading,
    error,
    refreshDashboard,
  };
};

  status: "all",
  stockStatus: "all",
  sort: "newest",
  page: 1,
  limit: 12,
};

const useAdminProducts = () => {
  const [products, setProducts] = useState([]);
  const [categories, setCategories] = useState([]);
  const [pagination, setPagination] = useState(null);

  const [filters, setFilters] = useState(INITIAL_FILTERS);
  const [searchInput, setSearchInput] = useState("");

  const [isLoading, setIsLoading] = useState(true);
  const [actionLoading, setActionLoading] = useState("");
  const [error, setError] = useState("");

  const fetchProducts = useCallback(async (signal) => {
    try {
      setError("");

      const params = {
        ...filters,
        category: filters.category || undefined,
      };

      const data = await adminProductService.getProducts(
        params,
        { signal }
      );

      setProducts(data.products);
      setCategories(data.categories);
      setPagination(data.pagination);
    } catch (requestError) {
      if (
        requestError.code === "ERR_CANCELED" ||
        signal?.aborted
      ) {
        return;
      }

      setError(
        requestError.response?.data?.message ||
          "Unable to load products. Please try again."
      );
    } finally {
      if (!signal?.aborted) {
        setIsLoading(false);
      }
    }
  }, [filters]);

  useEffect(() => {
    const controller = new AbortController();

    setIsLoading(true);
    fetchProducts(controller.signal);

    return () => controller.abort();
  }, [fetchProducts]);

  useEffect(() => {
    const timeout = setTimeout(() => {
      setFilters((current) => {
        if (current.search === searchInput.trim()) {
          return current;
        }

        return {
          ...current,
          search: searchInput.trim(),
          page: 1,
        };
      });
    }, 350);

    return () => clearTimeout(timeout);
  }, [searchInput]);

  const updateFilter = useCallback((name, value) => {
    setFilters((current) => ({
      ...current,
      [name]: value,
      page: 1,
    }));
  }, []);

  const changePage = useCallback((page) => {
    setFilters((current) => ({
      ...current,
      page,
    }));
  }, []);

  const refreshProducts = useCallback(() => {
    setIsLoading(true);
    fetchProducts();
  }, [fetchProducts]);

  const toggleProductStatus = useCallback(async (product) => {
    const productId = product._id;

    try {
      setActionLoading(productId);
      setError("");

      if (product.isActive) {
        await adminProductService.deactivateProduct(productId);
      } else {
        await adminProductService.activateProduct(productId);
      }

      await fetchProducts();
    } catch (requestError) {
      setError(
        requestError.response?.data?.message ||
          "Unable to update product status. Please try again."
      );
    } finally {
      setActionLoading("");
    }
  }, [fetchProducts]);

  return {
    products,
    categories,
    pagination,
    filters,
    searchInput,
    setSearchInput,
    updateFilter,
    changePage,
    refreshProducts,
    toggleProductStatus,
    isLoading,
    actionLoading,
    error,
  };
};

export default useAdminProducts;
