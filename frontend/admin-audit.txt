import { Navigate, Outlet } from "react-router-dom";

import useAccount from "../../account/hooks/useAccount";

const ProtectedAdminRoute = () => {
  const {
    currentUser,
    isAuthenticated,
    isLoading,
  } = useAccount();

  if (isLoading) {
    return (
      <div className="admin-auth-loading" role="status">
        Verifying administrator access...
      </div>
    );
  }

  if (!isAuthenticated) {
    return <Navigate to="/login" replace />;
  }

  if (currentUser?.role !== "admin") {
    return <Navigate to="/" replace />;
  }

  return <Outlet />;
};

export default ProtectedAdminRoute;

import { useCallback, useEffect, useState } from "react";

import adminCategoryService from "../services/adminCategoryService";

const useAdminCategories = () => {
  const [categories, setCategories] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [actionLoading, setActionLoading] = useState("");
  const [error, setError] = useState("");

  const fetchCategories = useCallback(async (signal) => {
    try {
      setError("");

      const data = await adminCategoryService.getCategories({ signal });

      setCategories(Array.isArray(data) ? data : []);
    } catch (requestError) {
      if (
        requestError.code === "ERR_CANCELED" ||
        signal?.aborted
      ) {
        return;
      }

      setError(
        requestError.response?.data?.message ||
          "Categories load nahi ho sakin."
      );
    } finally {
      if (!signal?.aborted) {
        setIsLoading(false);
      }
    }
  }, []);

  useEffect(() => {
    const controller = new AbortController();

    fetchCategories(controller.signal);

    return () => controller.abort();
  }, [fetchCategories]);

  const refreshCategories = useCallback(() => {
    setIsLoading(true);
    fetchCategories();
  }, [fetchCategories]);

  const toggleCategoryStatus = useCallback(async (category) => {
    try {
      setActionLoading(category._id);
      setError("");

      if (category.isActive) {
        await adminCategoryService.deactivateCategory(category._id);
      } else {
        await adminCategoryService.activateCategory(category._id);
      }

      await fetchCategories();
    } catch (requestError) {
      setError(
        requestError.response?.data?.message ||
          "Category status update nahi ho saka."
      );
    } finally {
      setActionLoading("");
    }
  }, [fetchCategories]);

  return {
    categories,
    isLoading,
    actionLoading,
    error,
    refreshCategories,
    toggleCategoryStatus,
  };
};

export default useAdminCategories;

import {
  useCallback,
  useEffect,
  useState,
} from "react";

import adminProductService from "../services/adminProductService";

const INITIAL_FILTERS = {
  search: "",
  category: "",
  status: "all",
  stockStatus: "all",
  sort: "newest",
  page: 1,
  limit: 12,
};

const useAdminProducts = () => {
  const [products, setProducts] = useState([]);
  const [categories, setCategories] = useState([]);
  const [pagination, setPagination] = useState(null);

  const [filters, setFilters] = useState(INITIAL_FILTERS);
  const [searchInput, setSearchInput] = useState("");

  const [isLoading, setIsLoading] = useState(true);
  const [actionLoading, setActionLoading] = useState("");
  const [error, setError] = useState("");

  const fetchProducts = useCallback(async (signal) => {
    try {
      setError("");

      const params = {
        ...filters,
        category: filters.category || undefined,
      };

      const data = await adminProductService.getProducts(
        params,
        { signal }
      );

      setProducts(data.products);
      setCategories(data.categories);
      setPagination(data.pagination);
    } catch (requestError) {
      if (
        requestError.code === "ERR_CANCELED" ||
        signal?.aborted
      ) {
        return;
      }

      setError(
        requestError.response?.data?.message ||
          "Unable to load products. Please try again."
      );
    } finally {
      if (!signal?.aborted) {
        setIsLoading(false);
      }
    }
  }, [filters]);

  useEffect(() => {
    const controller = new AbortController();

    setIsLoading(true);
    fetchProducts(controller.signal);

    return () => controller.abort();
  }, [fetchProducts]);

  useEffect(() => {
    const timeout = setTimeout(() => {
      setFilters((current) => {
        if (current.search === searchInput.trim()) {
          return current;
        }

        return {
          ...current,
          search: searchInput.trim(),
          page: 1,
        };
      });
    }, 350);

    return () => clearTimeout(timeout);
  }, [searchInput]);

  const updateFilter = useCallback((name, value) => {
    setFilters((current) => ({
      ...current,
      [name]: value,
      page: 1,
    }));
  }, []);

  const changePage = useCallback((page) => {
    setFilters((current) => ({
      ...current,
      page,
    }));
  }, []);

  const refreshProducts = useCallback(() => {
    setIsLoading(true);
    fetchProducts();
  }, [fetchProducts]);

  const toggleProductStatus = useCallback(async (product) => {
    const productId = product._id;

    try {
      setActionLoading(productId);
      setError("");

      if (product.isActive) {
        await adminProductService.deactivateProduct(productId);
      } else {
        await adminProductService.activateProduct(productId);
      }

      await fetchProducts();
    } catch (requestError) {
      setError(
        requestError.response?.data?.message ||
          "Unable to update product status. Please try again."
      );
    } finally {
      setActionLoading("");
    }
  }, [fetchProducts]);

  return {
    products,
    categories,
    pagination,
    filters,
    searchInput,
    setSearchInput,
    updateFilter,
    changePage,
    refreshProducts,
    toggleProductStatus,
    isLoading,
    actionLoading,
    error,
  };
};

export default useAdminProducts;

import {

import { Link, useNavigate, useParams } from "react-router-dom";

import adminCategoryService from "../services/adminCategoryService";

import "./admin-categories.css";

const INITIAL_FORM = {
  name: "",
  slug: "",
  description: "",
  image: "",
};

const slugify = (value) =>
  value
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");

const getErrorMessage = (error) =>
  error.response?.data?.message ||
  error.message ||
  "Something went wrong. Please try again.";

const AdminCategoryForm = () => {
  const { id } = useParams();
  const navigate = useNavigate();

  const isEditMode = Boolean(id);

  const [form, setForm] = useState(INITIAL_FORM);
  const [slugTouched, setSlugTouched] = useState(false);
  const [loading, setLoading] = useState(isEditMode);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    if (!isEditMode) return undefined;

    const controller = new AbortController();

    const loadCategory = async () => {
      try {
        setLoading(true);
        setError("");

        const categories = await adminCategoryService.getCategories({
          signal: controller.signal,
        });

        const category = categories.find(
          (item) => item._id === id
        );

        if (!category) {
          throw new Error("Requested category nahi mili.");
        }

        setForm({
          name: category.name ?? "",
          slug: category.slug ?? "",
          description: category.description ?? "",
          image: category.image ?? "",
        });

        setSlugTouched(true);
      } catch (requestError) {
        if (
          requestError.code === "ERR_CANCELED" ||
          controller.signal.aborted
        ) {
          return;
        }

        setError(getErrorMessage(requestError));
      } finally {
        if (!controller.signal.aborted) {
          setLoading(false);
        }
      }
    };

    loadCategory();

    return () => controller.abort();
  }, [id, isEditMode]);

        setIsLoading(false);

const STATUS_OPTIONS = [

  ["confirmed", "Confirmed"],
  ["processing", "Processing"],
  ["shipped", "Shipped"],
  ["delivered", "Delivered"],
  ["cancelled", "Cancelled"],

  ["", "All payment statuses"],
  ["pending", "Pending"],
  ["paid", "Paid"],
  ["failed", "Failed"],
  ["refunded", "Refunded"],

  ["", "All payment methods"],

const formatDate = (value) => {
  if (!value) return "â€”";

  const date = new Date(value);

    ? "â€”"

};

const getCustomerName = (order) =>
  order.customer?.fullName || "Guest customer";

  order.customer?.email || "No email available";

const getPaymentLabel = (status) =>
  ({
    pending: "Pending",
    paid: "Paid",
    failed: "Failed",
    refunded: "Refunded",

const AdminOrders = () => {

  const [showFilters, setShowFilters] = useState(false);

  const hasActiveFilters = useMemo(
    () =>
      Boolean(
        filters.status ||
        filters.paymentStatus ||
        filters.paymentMethod ||
        filters.from ||
        filters.to ||
        filters.search
      ),
    [filters]

    setSearchInput("");

    Object.entries({
      search: "",

  };

  return (
    <main className="admin-orders">
      <div className="admin-orders__container">
        <header className="admin-orders__header">
          <div className="admin-orders__heading">

              <ArrowLeft size={16} />
              Dashboard

              <Filter size={16} />
              Filters

          <form
            className="admin-orders__search"
            onSubmit={applySearch}
          >
            <Search size={18} />

            <input
              type="search"

import { useEffect, useRef, useState } from "react";
import {
  ArrowLeft,
  Image as ImageIcon,
  LoaderCircle,
  Save,
  ShieldCheck,
  Upload,
  X,
  Star,
} from "lucide-react";
import {
  Link,
  useNavigate,
  useParams,
} from "react-router-dom";

import adminProductService from "../services/adminProductService";

import "./admin-product-form.css";

const INITIAL_FORM = {
  sku: "",
  slug: "",
  name: "",
  category: "",
  categoryId: "",
  price: "",
  compareAtPrice: "",
  badge: "",
  description: "",
  stock: "0",
  isNewArrival: false,
  bestSeller: false,
  isActive: true,
};

const MAX_IMAGES = 8;
const MAX_IMAGE_SIZE = 5 * 1024 * 1024;

const ALLOWED_IMAGE_TYPES = [
  "image/jpeg",
  "image/png",
  "image/webp",
  "image/avif",
];

const slugify = (value) =>
  value
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");

const getErrorMessage = (error) =>
  error.response?.data?.message ||
  error.message ||
  "Something went wrong. Please try again.";

const createImageId = (file, index) =>
  `${Date.now()}-${index}-${file.name}`;

const AdminProductForm = () => {
  const { id } = useParams();
  const navigate = useNavigate();

  const isEditMode = Boolean(id);

  const fileInputRef = useRef(null);
  const objectUrlsRef = useRef(new Set());

  const [form, setForm] = useState(INITIAL_FORM);
  const [images, setImages] = useState([]);
  const [mainImageId, setMainImageId] = useState(null);

  const [categories, setCategories] = useState([]);
  const [categoriesLoading, setCategoriesLoading] = useState(true);

  const [slugTouched, setSlugTouched] = useState(false);
  const [loading, setLoading] = useState(isEditMode);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  // Load active categories.
  useEffect(() => {
    const controller = new AbortController();

    const loadCategories = async () => {
      try {
        setCategoriesLoading(true);

        const data = await adminProductService.getCategories({
          signal: controller.signal,
        });

        setCategories(Array.isArray(data) ? data : []);
      } catch (requestError) {
        if (
          requestError.code === "ERR_CANCELED" ||
          controller.signal.aborted
        ) {
          return;
        }

        setError(getErrorMessage(requestError));
      } finally {
        if (!controller.signal.aborted) {
          setCategoriesLoading(false);
        }
      }
    };

    loadCategories();

    return () => controller.abort();
  }, []);

  // Release temporary image previews.
  useEffect(() => {
    return () => {
      objectUrlsRef.current.forEach((url) => {
        URL.revokeObjectURL(url);
      });

      objectUrlsRef.current.clear();
    };
  }, []);

  // Load existing product in edit mode.
  useEffect(() => {
    if (!isEditMode) return undefined;

    const controller = new AbortController();

    const loadProduct = async () => {
      try {
        setLoading(true);
        setError("");

        const product = await adminProductService.getProductById(
          id,
          { signal: controller.signal }
        );

        setForm({
          sku: product.sku ?? "",
          slug: product.slug ?? "",
          name: product.name ?? "",
          category: product.category ?? "",
          categoryId: product.categoryId ?? "",
          price: String(product.price ?? ""),
          compareAtPrice:
            product.compareAtPrice == null
              ? ""
              : String(product.compareAtPrice),
          badge: product.badge ?? "",
          description: product.description ?? "",
          stock: String(product.stock ?? 0),
          isNewArrival: Boolean(product.isNewArrival),
          bestSeller: Boolean(product.bestSeller),
          isActive: Boolean(product.isActive),
        });

        const existingImages = Array.isArray(product.images)
          ? product.images.map((image, index) => ({
              id: image.publicId || `existing-${index}`,
              url: image.url,
              publicId: image.publicId,
              file: null,
              previewUrl: image.url,
              isExisting: true,
            }))
          : [];

        const initialImages = existingImages.length
          ? existingImages
          : product.image
            ? [
                {
                  id: "legacy-image",
                  url: product.image,
                  publicId: null,
                  file: null,
                  previewUrl: product.image,
                  isExisting: true,
                  isLegacy: true,
                },
              ]
            : [];

        setImages(initialImages);
        setMainImageId(initialImages[0]?.id ?? null);
        setSlugTouched(true);
      } catch (requestError) {
        if (
          requestError.code === "ERR_CANCELED" ||
          controller.signal.aborted
        ) {
          return;
        }

        setError(getErrorMessage(requestError));
      } finally {
        if (!controller.signal.aborted) {
          setLoading(false);
        }
      }
    };

    loadProduct();

    return () => controller.abort();
  }, [id, isEditMode]);

  const updateField = (name, value) => {
    setForm((current) => {
      const updated = { ...current, [name]: value };

      if (name === "name" && !slugTouched) {
        updated.slug = slugify(value);
      }

      return updated;
    });
  };

  const handleCategoryChange = (event) => {
    const categoryId = event.target.value;

    const selectedCategory = categories.find(
      (category) => category._id === categoryId
    );

    if (selectedCategory) {
      setForm((current) => ({
        ...current,
        categoryId: selectedCategory._id,
        category: selectedCategory.name,
      }));

      return;
    }

    // Preserve an existing legacy category.
    setForm((current) => ({
      ...current,
      categoryId: "",
    }));
  };

  const releaseImagePreview = (url) => {
    if (!url || !objectUrlsRef.current.has(url)) return;

    URL.revokeObjectURL(url);
    objectUrlsRef.current.delete(url);
  };

  const handleImageSelection = (event) => {
    const selectedFiles = Array.from(event.target.files || []);

    event.target.value = "";

    if (!selectedFiles.length) return;

    setError("");

    const availableSlots = MAX_IMAGES - images.length;

    if (availableSlots <= 0) {
      setError(`Maximum ${MAX_IMAGES} images allowed hain.`);
      return;
    }

    const acceptedFiles = [];
    const validationErrors = [];

    selectedFiles.forEach((file) => {
      if (!ALLOWED_IMAGE_TYPES.includes(file.type)) {
        validationErrors.push(
          `${file.name}: Unsupported image format.`
        );
        return;
      }

      if (file.size > MAX_IMAGE_SIZE) {
        validationErrors.push(
          `${file.name}: Image size 5MB se zyada hai.`
        );
        return;
      }

      acceptedFiles.push(file);
    });

    const filesToAdd = acceptedFiles.slice(0, availableSlots);

    if (acceptedFiles.length > availableSlots) {
      validationErrors.push(
        `Sirf ${availableSlots} additional images add ho sakti hain.`
      );
    }

    if (validationErrors.length) {
      setError(validationErrors.join(" "));
    }

    if (!filesToAdd.length) return;

    const newImages = filesToAdd.map((file, index) => {
      const previewUrl = URL.createObjectURL(file);

      objectUrlsRef.current.add(previewUrl);

      return {
        id: createImageId(file, index),
        url: "",
        publicId: null,
        file,
        previewUrl,
        isExisting: false,
      };
    });

    if (!images.some((image) => !image.isLegacy)) {
      setMainImageId(newImages[0].id);
    }

    setImages((current) => [...current, ...newImages]);
  };

  const handleRemoveImage = (imageId) => {
    const targetImage = images.find(
      (image) => image.id === imageId
    );

    if (!targetImage) return;

    releaseImagePreview(targetImage.previewUrl);

    const updatedImages = images.filter(
      (image) => image.id !== imageId
    );

    setImages(updatedImages);

    if (mainImageId === imageId) {
      setMainImageId(updatedImages[0]?.id ?? null);
    }
  };

  const handleSetMainImage = (imageId) => {
    const targetImage = images.find(
      (image) => image.id === imageId
    );

    if (!targetImage) return;

    if (
      targetImage.isLegacy &&
      images.some((image) => !image.isLegacy)
    ) {
      setError(
        "Cloudinary images mein se main image select karein."
      );
      return;
    }

    setMainImageId(imageId);
  };

  const handleSubmit = async (event) => {
    event.preventDefault();

    setError("");

    const mainImage =
      images.find((image) => image.id === mainImageId) ??
      images[0] ??
      null;

    if (!mainImage) {
      setError("Kam az kam aik product image select karein.");
      return;
    }

    if (mainImage.id !== mainImageId) {
      setMainImageId(mainImage.id);
    }

    const payload = {
      ...form,
      sku: form.sku.trim().toUpperCase(),
      slug: form.slug.trim(),
      name: form.name.trim(),
      category: form.category.trim(),
      price: Number(form.price),
      compareAtPrice:
        form.compareAtPrice.trim() === ""
          ? null
          : Number(form.compareAtPrice),
      badge: form.badge.trim() || null,
      description: form.description.trim(),
      stock: Number(form.stock),
      isNewArrival: Boolean(form.isNewArrival),
      bestSeller: Boolean(form.bestSeller),
      isActive: Boolean(form.isActive),
    };

    const selectedCategory = categories.find(
      (category) =>
        category._id === form.categoryId ||
        category.name === form.category
    );

    if (selectedCategory) {
      payload.categoryId = selectedCategory._id;
      payload.category = selectedCategory.name;
    } else if (form.categoryId) {
      setError(
        "Selected category available nahi hai. Dobara select karein."
      );
      return;
    } else {
      delete payload.categoryId;
    }

    if (
      !Number.isFinite(payload.price) ||
      payload.price < 0 ||
      !Number.isInteger(payload.stock) ||
      payload.stock < 0
    ) {
      setError(
        "Price aur stock valid non-negative numbers hone chahiye."
      );
      return;
    }

    if (
      payload.compareAtPrice !== null &&
      (!Number.isFinite(payload.compareAtPrice) ||
        payload.compareAtPrice <= payload.price)
    ) {
      setError(
        "Compare-at price selling price se zyada honi chahiye."
      );
      return;
    }

    try {
      setSaving(true);

      const newImages = images.filter(
        (image) => image.file
      );

      let uploadedImages = [];

      if (newImages.length) {
        uploadedImages =
          await adminProductService.uploadProductImages(
            newImages.map((image) => image.file)
          );
      }

      let uploadIndex = 0;

      const resolvedImages = images.map((image) => {
        if (!image.file) return image;

        const uploadedImage = uploadedImages[uploadIndex];

        uploadIndex += 1;

        if (!uploadedImage?.url || !uploadedImage?.publicId) {
          throw new Error(
            "Image upload response invalid hai. Dobara try karein."
          );
        }

        return {
          ...image,
          url: uploadedImage.url,
          publicId: uploadedImage.publicId,
        };
      });

      const orderedImages = [
        ...resolvedImages.filter(
          (image) => image.id === mainImageId
        ),
        ...resolvedImages.filter(
          (image) => image.id !== mainImageId
        ),
      ];

      const primaryImage = orderedImages[0];

      payload.image = primaryImage.url;

      const cloudinaryImages = orderedImages
        .filter((image) => image.publicId)
        .map(({ url, publicId }) => ({
          url,
          publicId,
        }));

      payload.images = cloudinaryImages;

      if (isEditMode) {
        delete payload.sku;

        await adminProductService.updateProduct(
          id,
          payload
        );
      } else {
        await adminProductService.createProduct(payload);
      }

      navigate("/admin/products", {
        replace: true,
        state: {
          message: isEditMode
            ? "Product updated successfully."
            : "Product created successfully.",
        },
      });
    } catch (requestError) {
      setError(getErrorMessage(requestError));
    } finally {
      setSaving(false);
    }
  };

  if (loading) {
    return (
      <main className="admin-product-form">
        <div className="admin-product-form__state">
          <LoaderCircle className="is-spinning" size={28} />
          <span>Loading product details...</span>
        </div>
      </main>
    );
  }

  return (
    <main className="admin-product-form">
      <div className="admin-product-form__container">
        <header className="admin-product-form__header">
          <div>
            <Link
              to="/admin/products"
              className="admin-product-form__back"
            >
              <ArrowLeft size={16} />
              Back to products
            </Link>

            <span className="admin-product-form__eyebrow">
              NEXAMART ADMINISTRATION
            </span>

            <h1>
              {isEditMode ? "Edit Product" : "Create Product"}
            </h1>

            <p>
              {isEditMode
                ? "Update your product information and catalog settings."
                : "Add a new product to your NexaMart catalog."}
            </p>
          </div>

          <div className="admin-product-form__security">
            <ShieldCheck size={17} />
            Authorized Administrator
          </div>
        </header>

        {error && (
          <div className="admin-product-form__error" role="alert">
            {error}
          </div>
        )}

        <form
          className="admin-product-form__layout"
          onSubmit={handleSubmit}
        >
          <section className="admin-product-form__panel">
            <div className="admin-product-form__section-heading">
              <h2>Product Information</h2>
              <p>Basic details displayed in your store catalog.</p>
            </div>

            <div className="admin-product-form__grid">
              <label className="admin-product-form__field">
                <span>Product Name *</span>
                <input
                  type="text"
                  value={form.name}
                  onChange={(event) =>
                    updateField("name", event.target.value)
                  }
                  minLength={2}
                  maxLength={160}
                  required
                  placeholder="Enter product name"
                />
              </label>

              <label className="admin-product-form__field">
                <span>SKU *</span>
                <input
                  type="text"
                  value={form.sku}
                  onChange={(event) =>
                    updateField(
                      "sku",
                      event.target.value.toUpperCase()
                    )
                  }
                  minLength={3}
                  maxLength={40}
                  required
                  readOnly={isEditMode}
                  placeholder="e.g. NM-009"
                />
                {isEditMode && (
                  <small>SKU cannot be changed after creation.</small>
                )}
              </label>

              <label className="admin-product-form__field">
                <span>Product Slug *</span>
                <input

    );

    return response.data.data.product;
  },

  async uploadProductImages(files) {
    if (!Array.isArray(files) || files.length === 0) {
      throw new Error("Kam az kam aik image select karein.");
    }

    const formData = new FormData();

    files.forEach((file) => {
      formData.append("images", file);
    });

    const response = await apiClient.post(
      "/products/images",
      formData
    );

    return response.data.data.images;
  },

  async createProduct(payload) {
    const response = await apiClient.post("/products", payload);

    return response.data.data.product ?? response.data.data;
  },

  async updateProduct(productId, payload) {
    const response = await apiClient.patch(
      `/products/${productId}`,
      payload
    );

    return response.data.data.product ?? response.data.data;
  },

  async activateProduct(productId) {
    const response = await apiClient.patch(
      `/admin/products/${productId}/activate`
    );

    return response.data.data.product;
  },

  async deactivateProduct(productId) {
