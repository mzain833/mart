import {
  
  Navigate,
  Route,
  Routes,
} from "react-router-dom";

import HomePage from "./pages/HomePage";
import ShopPage from "./features/products/pages/ShopPage";
import ProductDetailsPage from "./features/products/pages/ProductDetailsPage";
import CartPage from "./features/cart/pages/CartPage";
import WishlistPage from "./features/wishlist/pages/WishlistPage";
import CheckoutPage from "./features/checkout/pages/CheckoutPage";
import OrderConfirmationPage from "./features/checkout/pages/OrderConfirmationPage";

import LoginPage from "./features/account/pages/LoginPage";
import RegisterPage from "./features/account/pages/RegisterPage";
import ForgotPasswordPage from "./features/account/pages/ForgotPasswordPage";
import AccountDashboard from "./features/account/pages/AccountDashboard";
import ProfilePage from "./features/account/pages/ProfilePage";
import MyOrdersPage from "./features/account/pages/MyOrdersPage";
import ProtectedAccountRoute from "./features/account/components/ProtectedAccountRoute";

import OrderDetailsPage from "./features/orders/pages/OrderDetailsPage";

import ProtectedAdminRoute from "./features/admin/components/ProtectedAdminRoute";
import AdminDashboard from "./features/admin/pages/AdminDashboard";
import AdminProducts from "./features/admin/pages/AdminProducts";
import AdminProductForm from "./features/admin/pages/AdminProductForm";
import AdminCategories from "./features/admin/pages/AdminCategories";
import AdminCategoryForm from "./features/admin/pages/AdminCategoryForm";

function App() {
  return (
    <Routes>
      <Route path="/" element={<HomePage />} />

      <Route path="/shop" element={<ShopPage />} />
      <Route path="/categories" element={<ShopPage />} />

      <Route
        path="/new-arrivals"
        element={<ShopPage collection="new-arrivals" />}
      />

      <Route
        path="/best-sellers"
        element={<ShopPage collection="best-sellers" />}
      />

      <Route
        path="/product/:slug"
        element={<ProductDetailsPage />}
      />

      <Route path="/cart" element={<CartPage />} />
      <Route path="/wishlist" element={<WishlistPage />} />
      <Route path="/checkout" element={<CheckoutPage />} />

      <Route
        path="/order-confirmation/:orderId"
        element={<OrderConfirmationPage />}
      />

      <Route path="/login" element={<LoginPage />} />
      <Route path="/register" element={<RegisterPage />} />
      <Route path="/forgot-password" element={<ForgotPasswordPage />} />

      <Route element={<ProtectedAccountRoute />}>
        <Route path="/account" element={<AccountDashboard />} />
        <Route path="/account/profile" element={<ProfilePage />} />
        <Route path="/account/orders" element={<MyOrdersPage />} />

        <Route
          path="/account/orders/:orderId"
          element={<OrderDetailsPage />}
        />
      </Route>

      <Route element={<ProtectedAdminRoute />}>
        <Route path="/admin" element={<AdminDashboard />} />

        <Route path="/admin/products" element={<AdminProducts />} />

        <Route
          path="/admin/products/new"
          element={<AdminProductForm />}
        />

        <Route
          path="/admin/products/:id/edit"
          element={<AdminProductForm />}
        />
        <Route
  path="/admin/categories"
  element={<AdminCategories />}
/>

<Route
  path="/admin/categories/new"
  element={<AdminCategoryForm />}
/>

<Route
  path="/admin/categories/:id/edit"
  element={<AdminCategoryForm />}
/>
      </Route>

      <Route
        path="*"
        element={<Navigate to="/" replace />}
      />
    </Routes>
  );
}

export default App;

import { useEffect, useState } from "react";
import { Save } from "lucide-react";
import { validateProfile } from "../utils/accountValidation";

const ProfileForm = ({ user, onSave }) => {
  const [values, setValues] = useState({
    fullName: user.fullName || "",
    phone: user.phone || "",
  });

  const [errors, setErrors] = useState({});
  const [saving, setSaving] = useState(false);
  const [message, setMessage] = useState("");

  useEffect(() => {
    setValues({
      fullName: user.fullName || "",
      phone: user.phone || "",
    });
  }, [user]);

  const handleChange = (event) => {
    const { name, value } = event.target;

    setValues((previous) => ({ ...previous, [name]: value }));
    setErrors((previous) => ({ ...previous, [name]: "" }));
    setMessage("");
  };

  const handleSubmit = async (event) => {
    event.preventDefault();

    const validationErrors = validateProfile(values);
    setErrors(validationErrors);

    if (Object.keys(validationErrors).length) return;

    setSaving(true);

    try {
      await onSave(values);
      setMessage("Profile updated successfully.");
    } catch (error) {
      setMessage(error.message || "Unable to update profile.");
    } finally {
      setSaving(false);
    }
  };

  return (
    <form className="account-form" onSubmit={handleSubmit} noValidate>
      <div className="account-form-field">
        <label htmlFor="profile-name">Full Name</label>
        <input
          id="profile-name"
          name="fullName"
          value={values.fullName}
          onChange={handleChange}
          autoComplete="name"
          aria-invalid={Boolean(errors.fullName)}
        />
        {errors.fullName && (
          <span className="account-error">{errors.fullName}</span>
        )}
      </div>

      <div className="account-form-field">
        <label htmlFor="profile-email">Email Address</label>
        <input
          id="profile-email"
          type="email"
          value={user.email}
          disabled
        />
        <small className="account-field-note">
          Email changes require backend verification.
        </small>
      </div>

      <div className="account-form-field">
        <label htmlFor="profile-phone">Phone Number</label>
        <input
          id="profile-phone"
          name="phone"
          type="tel"
          value={values.phone}
          onChange={handleChange}
          autoComplete="tel"
          aria-invalid={Boolean(errors.phone)}
        />
        {errors.phone && (
          <span className="account-error">{errors.phone}</span>
        )}
      </div>

      {message && (
        <p className="account-form-message" role="status">
          {message}
        </p>
      )}

      <button
        type="submit"
        className="account-primary-button"

import apiClient from "../../../lib/apiClient";

let snapshot = {
  currentUser: null,
  isInitialized: false,
};

const listeners = new Set();
let initializationPromise = null;

const notify = () => {
  listeners.forEach((listener) => listener());
};

const updateSnapshot = (updates) => {
  snapshot = {
    ...snapshot,
    ...updates,
  };

  notify();
};

const getUser = (response) => response.data.data.user;

const accountService = {
  getSnapshot: () => snapshot,

  subscribe: (listener) => {
    listeners.add(listener);

    return () => listeners.delete(listener);
  },

  async initialize() {
    if (snapshot.isInitialized) {
      return snapshot.currentUser;
    }

    if (initializationPromise) {
      return initializationPromise;
    }

    initializationPromise = (async () => {
      try {
        const response = await apiClient.get("/auth/me");

        updateSnapshot({
          currentUser: getUser(response),
          isInitialized: true,
        });
      } catch {
        updateSnapshot({
          currentUser: null,
          isInitialized: true,
        });
      } finally {
        initializationPromise = null;
      }

      return snapshot.currentUser;
    })();

    return initializationPromise;
  },

  async register(values) {
    const response = await apiClient.post(
      "/auth/register",
      values
    );

    const user = getUser(response);

    updateSnapshot({
      currentUser: user,
      isInitialized: true,
    });

    return user;
  },

  async login(values) {
    const response = await apiClient.post(
      "/auth/login",
      values
    );

    const user = getUser(response);

    updateSnapshot({
      currentUser: user,
      isInitialized: true,
    });

    return user;
  },

  async logout() {
    try {
      await apiClient.post("/auth/logout");
    } finally {
      updateSnapshot({
        currentUser: null,
        isInitialized: true,
      });
    }
  },

  async updateProfile(values) {
    const response = await apiClient.patch(
      "/auth/profile",
      values
    );

    const user = getUser(response);

    updateSnapshot({
      currentUser: user,
    });

    return user;
  },

  async requestPasswordReset(email) {
    if (!email?.trim()) {
      throw new Error("Email address is required.");
    }

    throw new Error(
      "Password reset is not available yet."
    );
  },
};

export const validateRegistration = (values) => {
  const errors = {};

  if (!values.fullName.trim()) {
    errors.fullName = "Full name is required.";

    errors.fullName = "Name must contain at least 3 characters.";

  if (!values.email.trim()) {
    errors.email = "Email address is required.";
  } else if (!isValidEmail(values.email)) {

import adminProductService from "../services/adminProductService";

const INITIAL_FILTERS = {

  status: "all",

  sort: "newest",

const useAdminProducts = () => {
  const [products, setProducts] = useState([]);
  const [categories, setCategories] = useState([]);
  const [pagination, setPagination] = useState(null);

  const [filters, setFilters] = useState(INITIAL_FILTERS);
  const [searchInput, setSearchInput] = useState("");

  const [isLoading, setIsLoading] = useState(true);
  const [actionLoading, setActionLoading] = useState("");
  const [error, setError] = useState("");

  const fetchProducts = useCallback(async (signal) => {
    try {
      setError("");

      const params = {
        ...filters,
        category: filters.category || undefined,
      };

      const data = await adminProductService.getProducts(
        params,
        { signal }
      );

      setProducts(data.products);
      setCategories(data.categories);
      setPagination(data.pagination);
    } catch (requestError) {
      if (
        requestError.code === "ERR_CANCELED" ||
        signal?.aborted
      ) {
        return;
      }

      setError(
        requestError.response?.data?.message ||
          "Unable to load products. Please try again."
      );
    } finally {
      if (!signal?.aborted) {
        setIsLoading(false);
      }
    }
  }, [filters]);

  useEffect(() => {
    const controller = new AbortController();

    setIsLoading(true);
    fetchProducts(controller.signal);

    return () => controller.abort();
  }, [fetchProducts]);

  useEffect(() => {
    const timeout = setTimeout(() => {
      setFilters((current) => {
        if (current.search === searchInput.trim()) {
          return current;
        }

        return {
          ...current,
          search: searchInput.trim(),
          page: 1,
        };
      });
    }, 350);

    return () => clearTimeout(timeout);
  }, [searchInput]);

  const updateFilter = useCallback((name, value) => {
    setFilters((current) => ({
      ...current,
      [name]: value,
      page: 1,
    }));
  }, []);

  const changePage = useCallback((page) => {
    setFilters((current) => ({
      ...current,
      page,
    }));
  }, []);

  const refreshProducts = useCallback(() => {
    setIsLoading(true);
    fetchProducts();
  }, [fetchProducts]);

  const toggleProductStatus = useCallback(async (product) => {
    const productId = product._id;

    try {
      setActionLoading(productId);
      setError("");

      if (product.isActive) {
        await adminProductService.deactivateProduct(productId);
      } else {

      }

      await fetchProducts();
    } catch (requestError) {
      setError(
        requestError.response?.data?.message ||
          "Unable to update product status. Please try again."
      );
    } finally {
      setActionLoading("");
    }
  }, [fetchProducts]);

  return {
    products,
    categories,
    pagination,
    filters,
    searchInput,
    setSearchInput,
    updateFilter,
    changePage,
    refreshProducts,
    toggleProductStatus,

import {
  ArrowLeft,
  Image as ImageIcon,
  LoaderCircle,
  Save,
  ShieldCheck,
} from "lucide-react";
import { Link, useNavigate, useParams } from "react-router-dom";

import adminCategoryService from "../services/adminCategoryService";

import "./admin-categories.css";

const INITIAL_FORM = {
  name: "",
  slug: "",
  description: "",
  image: "",
};

const slugify = (value) =>
  value
    .toLowerCase()
    .trim()
    .replace(/[^a-z0-9]+/g, "-")
    .replace(/^-+|-+$/g, "");

const getErrorMessage = (error) =>
  error.response?.data?.message ||
  error.message ||
  "Something went wrong. Please try again.";

const AdminCategoryForm = () => {
  const { id } = useParams();
  const navigate = useNavigate();

  const isEditMode = Boolean(id);

  const [form, setForm] = useState(INITIAL_FORM);
  const [slugTouched, setSlugTouched] = useState(false);
  const [loading, setLoading] = useState(isEditMode);
  const [saving, setSaving] = useState(false);
  const [error, setError] = useState("");

  useEffect(() => {
    if (!isEditMode) return undefined;

    const controller = new AbortController();

    const loadCategory = async () => {
      try {
        setLoading(true);
        setError("");

        const categories = await adminCategoryService.getCategories({
          signal: controller.signal,
        });

        const category = categories.find(
          (item) => item._id === id
        );

        if (!category) {
          throw new Error("Requested category nahi mili.");
        }

        setForm({
          name: category.name ?? "",
          slug: category.slug ?? "",
          description: category.description ?? "",
          image: category.image ?? "",
        });

        setSlugTouched(true);
      } catch (requestError) {
        if (
          requestError.code === "ERR_CANCELED" ||
          controller.signal.aborted
        ) {
          return;
        }

        setError(getErrorMessage(requestError));
      } finally {
        if (!controller.signal.aborted) {
          setLoading(false);
        }
      }
    };

    loadCategory();

    return () => controller.abort();
  }, [id, isEditMode]);

  const updateField = (name, value) => {
    setForm((current) => {
      const updated = { ...current, [name]: value };

      if (name === "name" && !slugTouched) {
        updated.slug = slugify(value);
      }

      return updated;
    });
  };

  const handleSubmit = async (event) => {
    event.preventDefault();
    setError("");

    const payload = {
      name: form.name.trim(),
      slug: form.slug.trim(),
      description: form.description.trim(),
      image: form.image.trim() || null,
    };

    if (!payload.name || !payload.slug) {
      setError("Category name aur slug required hain.");
      return;
    }

    try {
      setSaving(true);

      if (isEditMode) {
        await adminCategoryService.updateCategory(id, payload);
      } else {
        await adminCategoryService.createCategory(payload);
      }

      navigate("/admin/categories", {
        replace: true,
        state: {
          message: isEditMode
            ? "Category updated successfully."
            : "Category created successfully.",
        },
      });
    } catch (requestError) {
      setError(getErrorMessage(requestError));
    } finally {
      setSaving(false);
    }
  };

  if (loading) {
    return (
      <main className="admin-category-form">
        <div className="admin-category-form__state">
          <LoaderCircle className="is-spinning" size={28} />
          <span>Loading category...</span>
        </div>
      </main>
    );
  }

  return (
    <main className="admin-category-form">
      <div className="admin-category-form__container">
        <header className="admin-category-form__header">
          <div>
            <Link
              to="/admin/categories"
              className="admin-category-form__back"
            >
              <ArrowLeft size={16} />
              Back to categories
            </Link>

            <span className="admin-category-form__eyebrow">
              NEXAMART ADMINISTRATION
            </span>

            <h1>{isEditMode ? "Edit Category" : "Create Category"}</h1>

            <p>
              {isEditMode
                ? "Update your category information."
                : "Add a new category to your store catalog."}
            </p>
          </div>

          <div className="admin-category-form__security">
            <ShieldCheck size={17} />
            Authorized Administrator
          </div>
        </header>

        {error && (
          <div className="admin-category-form__error" role="alert">
            {error}
          </div>
        )}

        <form
          className="admin-category-form__panel"
          onSubmit={handleSubmit}
        >
          <div className="admin-category-form__section-heading">
            <h2>Category Information</h2>
            <p>Enter the details used in your store catalog.</p>
          </div>

          <div className="admin-category-form__grid">
            <label className="admin-category-form__field">
              <span>Category Name *</span>
              <input
                type="text"
                value={form.name}
                onChange={(event) =>
                  updateField("name", event.target.value)
                }
                minLength={2}
                maxLength={80}
                required
                disabled={saving}
                placeholder="e.g. Men Shoes"
              />
            </label>

            <label className="admin-category-form__field">
              <span>Category Slug *</span>
              <input
                type="text"
                value={form.slug}
                onChange={(event) => {
                  setSlugTouched(true);
                  updateField("slug", slugify(event.target.value));
                }}
                minLength={2}
                maxLength={100}
                pattern="[a-z0-9]+(?:-[a-z0-9]+)*"
                required
                disabled={saving}
                placeholder="men-shoes"
              />
              <small>Category URL identifier.</small>
            </label>
          </div>

          <label className="admin-category-form__field">
            <span>Description</span>
            <textarea
              value={form.description}
              onChange={(event) =>
                updateField("description", event.target.value)
              }
              maxLength={500}
              rows={5}
              disabled={saving}
              placeholder="Write a short category description..."
            />

import { Minus, Plus, Trash2 } from "lucide-react";
import { Link } from "react-router-dom";

const formatPrice = (amount) =>
  new Intl.NumberFormat("en-PK", {
    style: "currency",
    currency: "PKR",
    maximumFractionDigits: 0,
  }).format(amount);

export default function CartItem({
  item,
  onUpdateQuantity,
  onRemove,
}) {
  const { product, quantity } = item;

  return (
    <article className="cart-item">
      <Link
        to={`/product/${product.slug}`}
        className="cart-item__image-link"
      >
        <img
          src={product.image}
          alt={product.name}
          className="cart-item__image"
          loading="lazy"
        />
      </Link>

      <div className="cart-item__details">
        <div>
          <span className="cart-item__category">
            {product.category}
          </span>

          <Link
            to={`/product/${product.slug}`}
            className="cart-item__name"
          >
            {product.name}
          </Link>

          <p className="cart-item__price">
            {formatPrice(product.price)}
          </p>
        </div>

        <div className="cart-item__actions">
          <div
            className="quantity-control"
            aria-label={`Quantity for ${product.name}`}
          >
            <button
              type="button"
              aria-label="Decrease quantity"
              disabled={quantity <= 1}
              onClick={() =>
                onUpdateQuantity(product.id, quantity - 1)
              }
            >
              <Minus size={15} />
            </button>

            <span aria-live="polite">{quantity}</span>

            <button
              type="button"
              aria-label="Increase quantity"
              disabled={quantity >= product.stock}
              onClick={() =>
                onUpdateQuantity(product.id, quantity + 1)
              }
            >
              <Plus size={15} />
            </button>
          </div>

          <button
            type="button"
            className="cart-item__remove"
            onClick={() => onRemove(product.id)}
          >
            <Trash2 size={16} />
            Remove
          </button>
        </div>
      </div>

      <div className="cart-item__total">
        {formatPrice(product.price * quantity)}
      </div>
    </article>
  );
}

import apiClient from "../../../lib/apiClient.js";

const getResponseData = (response) => response.data?.data;

const getMyOrders = async () => {
  const response = await apiClient.get("/orders/my-orders");

  return getResponseData(response)?.orders ?? [];
};

const getMyOrderById = async (orderId) => {
  const response = await apiClient.get(`/orders/${orderId}`);

  return getResponseData(response)?.order ?? null;
};

const getGuestOrderConfirmation = async (orderId, accessToken) => {
  const response = await apiClient.post(
    `/orders/guest/${orderId}/confirmation`,
    { accessToken }
  );

  return getResponseData(response)?.order ?? null;
};

const createOrder = async ({
  customer,
  items,
  deliveryMethod = "standard",
  paymentMethod = "cod",
}) => {
  const payload = {
    customer,
    items: items.map(({ product, quantity }) => ({
      sku: product.sku || product.id,
      quantity,
    })),
    deliveryMethod,
    paymentMethod,
  };

  const response = await apiClient.post("/orders", payload);

  const result = getResponseData(response);

  if (!result?.order) {
    throw new Error("Order could not be created. Please try again.");
  }

  return result;
};

const cancelOrder = async (orderId, reason) => {
  const response = await apiClient.patch(
    `/orders/${orderId}/cancel`,
    { reason }
  );

  return getResponseData(response)?.order ?? null;
};

const orderService = {
  createOrder,
  getMyOrders,
  getMyOrderById,
  getGuestOrderConfirmation,
  cancelOrder,
};

export default orderService;

export const formatOrderPrice = (amount) =>
  new Intl.NumberFormat("en-PK", {
    style: "currency",
    currency: "PKR",
    maximumFractionDigits: 0,
  }).format(Number(amount) || 0);

export const formatOrderDate = (date) =>
  new Intl.DateTimeFormat("en-PK", {
    dateStyle: "medium",
    timeStyle: "short",
  }).format(new Date(date));

export const getOrderStatusLabel = (status) => {
  const labels = {
    placed: "Order Placed",
    confirmed: "Confirmed",
    processing: "Processing",
    shipped: "Shipped",
    delivered: "Delivered",
    cancelled: "Cancelled",
  };

  return labels[status] || "Pending";
};

import { useEffect, useMemo, useState } from "react";
import { productService } from "../services/productService";
import {
  filterProducts,
  sortProducts,
} from "../utils/productUtils";

export default function useProducts({
  search = "",
  category = "all",
  maxPrice = 100000,
  minRating = 0,
  inStockOnly = false,
  sortBy = "featured",
  collection = "all",
} = {}) {
  const [products, setProducts] = useState([]);
  const [isLoading, setIsLoading] = useState(true);
  const [error, setError] = useState(null);

  useEffect(() => {
    let isMounted = true;

    async function loadProducts() {
      try {
        setIsLoading(true);
        setError(null);

        const data = await productService.getProducts();

        if (isMounted) {
          setProducts(data);
        }
      } catch {
        if (isMounted) {
          setError("Products load nahi ho sake. Dobara try karein.");
        }
      } finally {
        if (isMounted) {
          setIsLoading(false);
        }
      }
    }

    loadProducts();

    return () => {
      isMounted = false;
    };
  }, []);

  const filteredProducts = useMemo(() => {
    const filtered = filterProducts(products, {
      search,
      category,
      maxPrice,
      minRating,
      inStockOnly,
      collection,
    });

    return sortProducts(filtered, sortBy);
  }, [
    products,
    search,
    category,
    maxPrice,
    minRating,
    inStockOnly,
    sortBy,
    collection,
  ]);

  return {
    products: filteredProducts,
    totalProducts: products.length,
    isLoading,
    error,
  };
}

const normalizeProduct = (product) => ({

  };

  try {
    const response = await apiClient.get(

    );

    return normalizeProduct(response.data.data.product);
  } catch (error) {
    if (error.response?.status === 404) {
      return null;
    }

    throw error;
  }
};

export function formatPrice(price) {
  return new Intl.NumberFormat("en-PK", {
    style: "currency",
    currency: "PKR",
    maximumFractionDigits: 0,
  }).format(price);
}

export function getDiscountPercentage(price, compareAtPrice) {
  if (!compareAtPrice || compareAtPrice <= price) {
    return 0;
  }

  return Math.round(
    ((compareAtPrice - price) / compareAtPrice) * 100
  );
}

export function getProductCategories(products) {
  return [...new Set(products.map((product) => product.category))];
}

export function filterProducts(products, filters = {}) {
  const {
    search = "",
    category = "all",
    maxPrice = Infinity,
    minRating = 0,
    inStockOnly = false,
    collection = "all",
  } = filters;

  const normalizedSearch = search.trim().toLowerCase();

  return products.filter((product) => {
    const matchesSearch =
      !normalizedSearch ||
      product.name.toLowerCase().includes(normalizedSearch) ||
      product.category.toLowerCase().includes(normalizedSearch);

    const matchesCategory =
      category === "all" || product.category === category;

    const matchesPrice = product.price <= maxPrice;

    const matchesRating = product.rating >= minRating;

    const matchesStock =
      !inStockOnly || product.stock > 0;

    const matchesCollection =
      collection === "all" ||
      (collection === "new-arrivals" && product.isNew) ||
      (collection === "best-sellers" && product.bestSeller);

    return (
      matchesSearch &&
      matchesCategory &&
      matchesPrice &&
      matchesRating &&
      matchesStock &&
      matchesCollection
    );
  });
}

export function sortProducts(products, sortBy = "featured") {
  const sorted = [...products];

  switch (sortBy) {
    case "price-low":
      return sorted.sort((a, b) => a.price - b.price);

    case "price-high":
      return sorted.sort((a, b) => b.price - a.price);

    case "rating":
      return sorted.sort((a, b) => b.rating - a.rating);

    case "newest":
      return sorted.sort(
        (a, b) => Number(b.isNew) - Number(a.isNew)
      );

    default:
      return sorted.sort(
        (a, b) => Number(b.bestSeller) - Number(a.bestSeller)
      );
  }
}

export default function WishlistItem({
  product,
  onRemove,
  onAddToCart,
}) {
  const isOutOfStock = Number(product.stock) <= 0;

  return (
    <article className="wishlist-item">
      <Link
        to={`/product/${product.slug}`}
        className="wishlist-item__image-link"
      >
        <img
          src={product.image}
          alt={product.name}
          className="wishlist-item__image"
          loading="lazy"
        />
      </Link>

      <div className="wishlist-item__content">
        <span className="wishlist-item__category">
          {product.category}
        </span>

        <Link
          to={`/product/${product.slug}`}
          className="wishlist-item__name"
        >
          {product.name}
        </Link>

        <strong className="wishlist-item__price">
          {formatPrice(product.price)}
        </strong>

        <span
          className={
            isOutOfStock
              ? "wishlist-item__stock is-out"
              : "wishlist-item__stock"
          }
        >
          {isOutOfStock ? "Out of stock" : "In stock"}
        </span>

        <div className="wishlist-item__actions">
          <button
            type="button"
            className="wishlist-item__add"
            disabled={isOutOfStock}
            onClick={() => onAddToCart(product)}
          >
            <ShoppingBag size={16} />
            Add to Cart
          </button>

          <button
            type="button"

      <Heart
        className="wishlist-item__heart"

      />
    </article>
  );
}
