import mongoose from "mongoose";

import Order from "../../models/order/Order.model.js";
import Product from "../../models/product/Product.model.js";
import User from "../../models/user/User.model.js";
import ApiError from "../../utils/ApiError.js";
import { HTTP_STATUS } from "../../constants/index.js";

const LOW_STOCK_THRESHOLD = 5;
const RECENT_ORDERS_LIMIT = 5;
const LOW_STOCK_LIMIT = 5;

const DEFAULT_PAGE = 1;
const DEFAULT_LIMIT = 12;
const MAX_LIMIT = 100;
const MAX_SEARCH_LENGTH = 100;

const SORT_OPTIONS = Object.freeze({
  newest: { createdAt: -1, _id: -1 },
  oldest: { createdAt: 1, _id: 1 },
  price_asc: { price: 1, _id: 1 },
  price_desc: { price: -1, _id: -1 },
  name_asc: { name: 1, _id: 1 },
  name_desc: { name: -1, _id: -1 },
});

const throwBadRequest = (message) => {
  throw new ApiError(HTTP_STATUS.BAD_REQUEST, message);
};

const escapeRegex = (value) =>
  value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

const parseInteger = (value, field, min, max) => {
  if (
    typeof value !== "string" ||
    !/^\d+$/.test(value)
  ) {
    throwBadRequest(
      `${field} must be an integer between ${min} and ${max}.`
    );
  }

  const parsed = Number(value);

  if (!Number.isSafeInteger(parsed) || parsed < min || parsed > max) {
    throwBadRequest(
      `${field} must be an integer between ${min} and ${max}.`
    );
  }

  return parsed;
};

const normalizeAdminProductQuery = (query = {}) => {
  const allowedFields = new Set([
    "search",
    "category",
    "status",
    "stockStatus",
    "sort",
    "page",
    "limit",
  ]);

  for (const field of Object.keys(query)) {
    if (!allowedFields.has(field)) {
      throwBadRequest(`Unsupported query parameter: ${field}.`);
    }
  }

  if (
    query.search !== undefined &&
    typeof query.search !== "string"
  ) {
    throwBadRequest("Search must be a valid string.");
  }

  if (
    query.category !== undefined &&
    typeof query.category !== "string"
  ) {
    throwBadRequest("Category must be a valid string.");
  }

  const search = query.search?.trim();
  const category = query.category?.trim();

  if (search && search.length > MAX_SEARCH_LENGTH) {
    throwBadRequest("Search query is too long.");
  }

  if (query.category !== undefined && !category) {
    throwBadRequest("Category cannot be empty.");
  }

  const status = query.status ?? "all";

  if (!["all", "active", "inactive"].includes(status)) {
    throwBadRequest("Invalid product status filter.");
  }

  const stockStatus = query.stockStatus ?? "all";

  if (
    !["all", "in_stock", "low_stock", "out_of_stock"].includes(
      stockStatus
    )
  ) {
    throwBadRequest("Invalid stock status filter.");
  }

  const sort = query.sort ?? "newest";

  if (
    typeof sort !== "string" ||
    !Object.hasOwn(SORT_OPTIONS, sort)
  ) {
    throwBadRequest("Invalid product sorting option.");
  }

  const page =
    query.page === undefined
      ? DEFAULT_PAGE
      : parseInteger(query.page, "Page", 1, 1000000);

  const limit =
    query.limit === undefined
      ? DEFAULT_LIMIT
      : parseInteger(query.limit, "Limit", 1, MAX_LIMIT);

  return { search, category, status, stockStatus, sort, page, limit };
};

const getDashboardStats = async () => {
  const [
    totalOrders,
    totalProducts,
    totalCustomers,
    activeOrders,
    revenueResult,
    orderValueResult,
    lowStockCount,
    recentOrders,
    lowStockProducts,
  ] = await Promise.all([
    Order.countDocuments(),

    Product.countDocuments({ isActive: true }),

    User.countDocuments({ role: "customer" }),

    Order.countDocuments({
      status: { $in: ["placed", "confirmed", "processing", "shipped"] },
    }),

    Order.aggregate([
      {
        $match: {
          paymentStatus: "paid",
          status: { $ne: "cancelled" },
        },
      },
      { $group: { _id: null, total: { $sum: "$total" } } },
    ]),

    Order.aggregate([
      { $match: { status: { $ne: "cancelled" } } },
      { $group: { _id: null, total: { $sum: "$total" } } },
    ]),

    Product.countDocuments({
      isActive: true,
      stock: { $lte: LOW_STOCK_THRESHOLD },
    }),

    Order.find()
      .select(
        "orderNumber customer.fullName total status paymentStatus createdAt"
      )
      .sort({ createdAt: -1, _id: -1 })
      .limit(RECENT_ORDERS_LIMIT)
      .lean(),

    Product.find({
      isActive: true,
      stock: { $lte: LOW_STOCK_THRESHOLD },
    })
      .select("name sku price stock image category")
      .sort({ stock: 1, _id: 1 })
      .limit(LOW_STOCK_LIMIT)
      .lean(),
  ]);

  return {
    stats: {
      totalRevenue: revenueResult[0]?.total ?? 0,
      totalOrderValue: orderValueResult[0]?.total ?? 0,
      totalOrders,
      activeOrders,
      totalProducts,
      totalCustomers,
      lowStockCount,
    },
    recentOrders,
    lowStockProducts,
  };
};

const getAdminProducts = async (query = {}) => {
  const { search, category, status, stockStatus, sort, page, limit } =
    normalizeAdminProductQuery(query);

  const filter = {};

  if (status === "active") filter.isActive = true;
  if (status === "inactive") filter.isActive = false;

  if (search) {
    const searchRegex = new RegExp(escapeRegex(search), "i");

    filter.$or = [
      { name: searchRegex },
      { sku: searchRegex },
    ];
  }

  if (category) {
    filter.category = category;
  }

  if (stockStatus === "in_stock") {
    filter.stock = { $gt: LOW_STOCK_THRESHOLD };
  }

  if (stockStatus === "low_stock") {
    filter.stock = { $gt: 0, $lte: LOW_STOCK_THRESHOLD };
  }

  if (stockStatus === "out_of_stock") {
    filter.stock = 0;
  }

  const skip = (page - 1) * limit;

  const [products, total, categories] = await Promise.all([
    Product.find(filter)
      .sort(SORT_OPTIONS[sort])
      .skip(skip)
      .limit(limit)
      .lean(),

    Product.countDocuments(filter),

    Product.distinct("category"),
  ]);

  return {
    products,
    categories: categories.sort((a, b) => a.localeCompare(b)),
    pagination: {
      page,
      limit,
      total,
      pages: Math.ceil(total / limit),
      hasNextPage: page * limit < total,
      hasPreviousPage: page > 1,
    },
  };
};

const getAdminProductById = async (productId) => {
  if (!mongoose.isValidObjectId(productId)) {
    throwBadRequest("Invalid product ID.");
  }

  const product = await Product.findById(productId).lean();

  if (!product) {
    throw new ApiError(
      HTTP_STATUS.NOT_FOUND,
      "Product not found."
    );
  }

  return product;
};

const updateProductStatus = async (productId, isActive) => {
  if (!mongoose.isValidObjectId(productId)) {
    throwBadRequest("Invalid product ID.");
  }

  const product = await Product.findOneAndUpdate(
    { _id: productId, isActive: !isActive },
    { $set: { isActive } },
    { new: true, runValidators: true }
  ).lean();

  if (!product) {
    const exists = await Product.exists({ _id: productId });

    if (!exists) {
      throw new ApiError(
        HTTP_STATUS.NOT_FOUND,
        "Product not found."
      );
    }

    throw new ApiError(
      HTTP_STATUS.CONFLICT,
      `Product is already ${isActive ? "active" : "inactive"}.`
    );
  }

  return product;
};

const activateProduct = (productId) =>
  updateProductStatus(productId, true);

const deactivateProduct = (productId) =>
  updateProductStatus(productId, false);

const adminService = {
  getDashboardStats,
  getAdminProducts,
  getAdminProductById,
  activateProduct,
  deactivateProduct,
};

export default adminService;

import { randomBytes, randomUUID, createHash } from "node:crypto";

import mongoose from "mongoose";

import Order from "../../models/order/Order.model.js";
import Product from "../../models/product/Product.model.js";

import ApiError from "../../utils/ApiError.js";
import { HTTP_STATUS } from "../../constants/index.js";

const MAX_ORDER_QUANTITY = 100;
const GUEST_TOKEN_BYTES = 32;

const generateOrderNumber = () =>
  `NEXA-${Date.now().toString().slice(-8)}-${randomUUID()
    .replace(/-/g, "")
    .slice(0, 8)
    .toUpperCase()}`;

const generateGuestAccessToken = () =>
  randomBytes(GUEST_TOKEN_BYTES).toString("hex");

const hashGuestAccessToken = (token) =>
  createHash("sha256").update(token).digest("hex");

const normalizeEmail = (email) =>
  String(email || "").trim().toLowerCase();

const sanitizeOrder = (order) => {
  if (!order) return null;

  const safeOrder = order.toObject
    ? order.toObject()
    : { ...order };

  delete safeOrder.guestAccessTokenHash;

  return safeOrder;
};

const validateOrderId = (orderId) => {
  if (!mongoose.isValidObjectId(orderId)) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid order ID."
    );
  }
};

const createOrder = async ({
  customer,
  items,
  deliveryMethod,
  paymentMethod,
  userId = null,
}) => {
  if (!customer || !Array.isArray(items) || items.length === 0) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Required order information is missing."
    );
  }

  if (paymentMethod !== "cod") {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Currently, only Cash on Delivery is available."
    );
  }

  if (!["standard", "express"].includes(deliveryMethod)) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid delivery method."
    );
  }

  if (items.length > 50) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "An order cannot contain more than 50 products."
    );
  }

  const normalizedItems = items.map((item) => ({
    sku: String(item.sku || "").trim().toUpperCase(),
    quantity: Number(item.quantity),
  }));

  const uniqueSkus = new Set(
    normalizedItems.map((item) => item.sku)
  );

  if (
    uniqueSkus.size !== normalizedItems.length ||
    normalizedItems.some(
      (item) =>
        !item.sku ||
        !Number.isInteger(item.quantity) ||
        item.quantity < 1 ||
        item.quantity > MAX_ORDER_QUANTITY
    )
  ) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Invalid or duplicate products in your order."
    );
  }

  const guestAccessToken = userId
    ? null
    : generateGuestAccessToken();

  const guestAccessTokenHash = guestAccessToken
    ? hashGuestAccessToken(guestAccessToken)
    : null;

  const session = await mongoose.startSession();

  let createdOrder;

  try {
    await session.withTransaction(async () => {
      const products = await Product.find({
        sku: { $in: normalizedItems.map((item) => item.sku) },
        isActive: true,
      }).session(session);

      if (products.length !== normalizedItems.length) {
        throw new ApiError(
          HTTP_STATUS.NOT_FOUND,
          "One or more products are unavailable."
        );
      }

      const productMap = new Map(
        products.map((product) => [product.sku, product])
      );

      const orderItems = [];
      let subtotal = 0;

      for (const item of normalizedItems) {
        const product = productMap.get(item.sku);

        if (!product) {
          throw new ApiError(
            HTTP_STATUS.NOT_FOUND,
            `Product ${item.sku} was not found.`
          );
        }

        if (product.stock < item.quantity) {
          throw new ApiError(
            HTTP_STATUS.CONFLICT,
            `Insufficient stock for ${product.name}.`
          );
        }

        const unitPrice = Number(product.price);
        const lineTotal = unitPrice * item.quantity;

        if (
          !Number.isFinite(unitPrice) ||
          unitPrice < 0 ||
          !Number.isFinite(lineTotal)
        ) {
          throw new ApiError(
            HTTP_STATUS.INTERNAL_SERVER_ERROR,
            "Invalid product pricing configuration."
          );
        }

        subtotal += lineTotal;

        orderItems.push({
          product: {
            productId: product._id,
            sku: product.sku,
            name: product.name,
            slug: product.slug,
            image: product.image,
          },
          quantity: item.quantity,
          unitPrice,
          lineTotal,
        });
      }

      for (const item of normalizedItems) {
        const product = productMap.get(item.sku);

        const stockUpdate = await Product.updateOne(
          {
            _id: product._id,
            isActive: true,
            stock: { $gte: item.quantity },
          },
          {
            $inc: { stock: -item.quantity },
          },
          { session }
        );

        if (stockUpdate.modifiedCount !== 1) {
          throw new ApiError(
            HTTP_STATUS.CONFLICT,
            `Stock changed for ${product.name}. Please try again.`
          );
        }
      }

      const orderData = {
        orderNumber: generateOrderNumber(),
        user: userId || null,
        guestAccessTokenHash,
        customer: {
          fullName: customer.fullName.trim(),
          email: normalizeEmail(customer.email),
          phone: customer.phone.trim(),
          addressLine1: customer.addressLine1.trim(),
          addressLine2: customer.addressLine2?.trim() || "",
          city: customer.city.trim(),
          province: customer.province.trim(),
          postalCode: customer.postalCode.trim(),
        },
        items: orderItems,
        subtotal,
        deliveryFee: null,
        total: subtotal,
        deliveryMethod,
        paymentMethod: "cod",
        paymentStatus: "pending",
        status: "placed",
        statusHistory: [
          {
            status: "placed",
            timestamp: new Date(),
            note: "Your order has been successfully placed.",
          },
        ],
      };

      const [order] = await Order.create(
        [orderData],
        { session }
      );

      createdOrder = order;
    });
  } finally {
    await session.endSession();
  }

  return {
    order: sanitizeOrder(createdOrder),
    guestAccessToken,
  };
};

const getGuestOrderConfirmation = async (
  orderId,
  accessToken
) => {
  validateOrderId(orderId);

  if (
    typeof accessToken !== "string" ||
    !/^[a-f\d]{64}$/i.test(accessToken)
  ) {
    throw new ApiError(
      HTTP_STATUS.UNAUTHORIZED,
      "Invalid order access credentials."
    );
  }

  const tokenHash = hashGuestAccessToken(accessToken);

  const order = await Order.findOne({
    _id: orderId,
    user: null,
    guestAccessTokenHash: tokenHash,
  })
    .select("+guestAccessTokenHash")
    .lean();

  if (!order) {
    throw new ApiError(
      HTTP_STATUS.NOT_FOUND,
      "Order not found or access denied."
    );
  }

  delete order.guestAccessTokenHash;

  return order;
};

const cancelOrder = async (orderId, userId, reason) => {
  validateOrderId(orderId);

  const normalizedReason = String(reason || "").trim();

  if (
    normalizedReason.length < 5 ||
    normalizedReason.length > 500
  ) {
    throw new ApiError(
      HTTP_STATUS.BAD_REQUEST,
      "Cancellation reason must contain 5 to 500 characters."
    );
  }

  const session = await mongoose.startSession();

  let cancelledOrder;

  try {
    await session.withTransaction(async () => {
      const order = await Order.findOne({
        _id: orderId,
        user: userId,
      }).session(session);

      if (!order) {
        throw new ApiError(
          HTTP_STATUS.NOT_FOUND,
          "Order not found."
        );
      }

      if (
        !["placed", "confirmed", "processing"].includes(
          order.status
        )
      ) {
        throw new ApiError(
          HTTP_STATUS.BAD_REQUEST,
          "This order is no longer eligible for cancellation."
        );
      }

      for (const item of order.items) {
        const stockUpdate = await Product.updateOne(
          { _id: item.product.productId },
          { $inc: { stock: item.quantity } },
          { session }
        );

        if (stockUpdate.matchedCount !== 1) {
          throw new ApiError(
            HTTP_STATUS.CONFLICT,
            "Unable to restore product inventory."
          );
        }
      }

      order.status = "cancelled";
      order.cancellationReason = normalizedReason;

      order.statusHistory.push({
        status: "cancelled",
        timestamp: new Date(),
        note: normalizedReason,
      });

      await order.save({ session });

      cancelledOrder = order;
    });
  } finally {
    await session.endSession();
  }

  return sanitizeOrder(cancelledOrder);
};

  getCustomerOrderById,
  getGuestOrderConfirmation,
  cancelOrder,
};

export default orderService;
