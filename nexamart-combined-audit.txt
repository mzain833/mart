import {
  AlertTriangle,
  Edit3,
  LoaderCircle,
  Package,
  Power,
  PowerOff,
} from "lucide-react";
import { Link } from "react-router-dom";

const formatCurrency = (value = 0) =>
  new Intl.NumberFormat("en-PK", {
    style: "currency",
    currency: "PKR",
    maximumFractionDigits: 0,
  }).format(Number(value) || 0);

const getStockStatus = (value) => {
  const stock = Math.max(0, Number(value) || 0);

  if (stock === 0) {
    return { label: "Out of stock", className: "out" };
  }

  if (stock <= 5) {
    return { label: "Low stock", className: "low" };
  }

  return { label: "In stock", className: "available" };
};

const AdminProductTable = ({
  products = [],
  isLoading,
  actionLoading,
  onToggleStatus,
}) => {
  if (isLoading && products.length === 0) {
    return (
      <div className="admin-products__state" role="status">
        <LoaderCircle size={28} className="is-spinning" />
        <span>Loading products...</span>
      </div>
    );
  }

  if (!isLoading && products.length === 0) {
    return (
      <div className="admin-products__state">
        <Package size={34} />
        <strong>No products found</strong>
        <span>Try changing your search or filters.</span>
      </div>
    );
  }

  return (
    <div className="admin-products__table-wrap">
      <table className="admin-products__table">
        <thead>
          <tr>
            <th>Product</th>
            <th>SKU</th>
            <th>Category</th>
            <th>Price</th>
            <th>Inventory</th>
            <th>Status</th>
            <th>Actions</th>
          </tr>
        </thead>

        <tbody>
          {products.map((product) => {
            const stock = getStockStatus(product.stock);
            const isUpdating = actionLoading === product._id;

            return (
              <tr key={product._id}>
                <td>
                  <div className="admin-products__product">
                    {product.image ? (
                      <img
                        src={product.image}
                        alt={product.name}
                        loading="lazy"
                      />
                    ) : (
                      <span className="admin-products__image-placeholder">
                        <Package size={19} />
                      </span>
                    )}

                    <div>
                      <strong>{product.name}</strong>
                      <span>{product.slug}</span>
                    </div>
                  </div>
                </td>

                <td className="admin-products__sku">
                  {product.sku || "â€”"}
                </td>

                <td>{product.category || "Uncategorized"}</td>

                <td className="admin-products__price">
                  {formatCurrency(product.price)}
                </td>

                <td>
                  <span
                    className={`admin-products__stock admin-products__stock--${stock.className}`}
                  >
                    {stock.className === "out" && (
                      <AlertTriangle size={13} />
                    )}
                    {Number(product.stock) || 0} Â· {stock.label}
                  </span>
                </td>

                <td>
                  <span
                    className={`admin-products__status ${
                      product.isActive ? "is-active" : "is-inactive"
                    }`}
                  >
                    <span />
                    {product.isActive ? "Active" : "Inactive"}
                  </span>
                </td>

                <td>
                  <div className="admin-products__actions">
                    <Link
                      to={`/admin/products/${product._id}/edit`}
                      className="admin-products__edit"
                      aria-label={`Edit ${product.name}`}
                    >
                      <Edit3 size={15} />
                      <span>Edit</span>
                    </Link>

                    <button
                      type="button"
                      className={`admin-products__action ${
                        product.isActive
                          ? "is-deactivate"
                          : "is-activate"
                      }`}
                      onClick={() => onToggleStatus(product)}
                      disabled={Boolean(actionLoading)}
                      aria-busy={isUpdating}
                    >
                      {isUpdating ? (
                        <LoaderCircle
                          size={15}
                          className="is-spinning"
                        />
                      ) : product.isActive ? (
                        <PowerOff size={15} />
                      ) : (
                        <Power size={15} />
                      )}

                      <span>
                        {product.isActive ? "Deactivate" : "Activate"}
                      </span>
                    </button>
                  </div>
                </td>
              </tr>
            );
          })}
        </tbody>
      </table>
    </div>
  );
};

export default AdminProductTable;

  Eye,
  Filter,
  LoaderCircle,
  RefreshCw,
  Search,
  ShieldCheck,
  Users,
  X,
} from "lucide-react";

import useAdminCustomers from "../hooks/useAdminCustomers";
import "./admin-customers.css";

const SORT_OPTIONS = [
  ["newest", "Newest first"],
  ["oldest", "Oldest first"],
  ["name_asc", "Name A-Z"],
  ["name_desc", "Name Z-A"],
];

const formatDate = (value) => {
  if (!value) return "â€”";

  const date = new Date(value);

  return Number.isNaN(date.getTime())
    ? "â€”"
    : new Intl.DateTimeFormat("en-PK", {
        dateStyle: "medium",
      }).format(date);
};

const formatPrice = (amount) =>
  new Intl.NumberFormat("en-PK", {
    style: "currency",
    currency: "PKR",
    maximumFractionDigits: 0,
  }).format(Number(amount) || 0);

const AdminCustomers = () => {
  const {
    customers,
    filters,
    searchInput,
    setSearchInput,
    pagination,
    isLoading,
    error,
    updateFilter,
    applySearch,
    changePage,
    refreshCustomers,
  } = useAdminCustomers();

  const [showFilters, setShowFilters] = useState(false);

  const hasActiveFilters = useMemo(
    () => Boolean(filters.status || filters.search),
    [filters.status, filters.search]
  );

  const clearFilters = () => {
    setSearchInput("");

    updateFilter("search", "");
    updateFilter("status", "");
    updateFilter("sort", "newest");
  };

  return (

<Link

>
  <Eye size={15} />
  <span>View</span>
</Link>

            <span className="admin-customers__eyebrow">
              NEXAMART ADMINISTRATION
            </span>

            <h1>Customer Management</h1>

            <p>
              Manage customer accounts, monitor activity and review
              customer relationships.
            </p>
          </div>

          <div className="admin-customers__header-actions">

            </span>

              <RefreshCw

              />
              Refresh
            </button>

                : "No pages available"}
            </small>
          </div>
        </section>

        <section className="admin-customers__panel">
          <div className="admin-customers__panel-header">
            <div>

            </div>

            <button
              type="button"
              className="admin-customers__filter-toggle"

              <Filter size={16} />

                </tr>

                    </td>

                ) : (

                    <tr key={customer._id}>

                            {(customer.fullName || "C")
                              .charAt(0)
                              .toUpperCase()}
                          </span>

                          <div>
                            <strong>{customer.fullName}</strong>
                            <span>{customer.email}</span>

                        </div>

                      <td>{customer.phone || "â€”"}</td>

                      <td>
                        <strong>
                          {customer.orderStats?.totalOrders ?? 0}

                      <td>
                        <span

                      </td>

                      <td>{formatDate(customer.createdAt)}</td>

export default AdminCustomers;

import mongoose from "mongoose";

import Order from "../../models/order/Order.model.js";
import Product from "../../models/product/Product.model.js";
import Category from "../../models/category/Category.model.js";
import User from "../../models/user/User.model.js";
import ApiError from "../../utils/ApiError.js";
import { HTTP_STATUS } from "../../constants/index.js";

const LOW_STOCK_THRESHOLD = 5;
const RECENT_ORDERS_LIMIT = 5;
const LOW_STOCK_LIMIT = 5;

const DEFAULT_PAGE = 1;
const DEFAULT_LIMIT = 12;
const MAX_LIMIT = 100;
const MAX_SEARCH_LENGTH = 100;

const SORT_OPTIONS = Object.freeze({
  newest: { createdAt: -1, _id: -1 },
  oldest: { createdAt: 1, _id: 1 },
  price_asc: { price: 1, _id: 1 },
  price_desc: { price: -1, _id: -1 },
  name_asc: { name: 1, _id: 1 },
  name_desc: { name: -1, _id: -1 },
});

const ACTIVE_ORDER_STATUSES = [
  "placed",
  "confirmed",
  "processing",
  "shipped",
];

const throwBadRequest = (message) => {
  throw new ApiError(HTTP_STATUS.BAD_REQUEST, message);
};

const escapeRegex = (value) =>
  value.replace(/[.*+?^${}()|[\]\\]/g, "\\$&");

const parseInteger = (value, field, min, max) => {
  if (
    typeof value !== "string" ||
    !/^\\d+$/.test(value)
  ) {
    throwBadRequest(
      `${field} must be an integer between ${min} and ${max}.`
    );
  }

  const parsed = Number(value);

  if (!Number.isSafeInteger(parsed) || parsed < min || parsed > max) {
    throwBadRequest(
      `${field} must be an integer between ${min} and ${max}.`
    );
  }

  return parsed;
};

const normalizeAdminProductQuery = (query = {}) => {
  const allowedFields = new Set([
    "search",
    "category",
    "status",
    "stockStatus",
    "sort",
    "page",
    "limit",
  ]);

  for (const field of Object.keys(query)) {
    if (!allowedFields.has(field)) {
      throwBadRequest(`Unsupported query parameter: ${field}.`);
    }
  }

  if (
    query.search !== undefined &&
    typeof query.search !== "string"
  ) {
    throwBadRequest("Search must be a valid string.");
  }

  if (
    query.category !== undefined &&
    typeof query.category !== "string"
  ) {
    throwBadRequest("Category must be a valid string.");
  }

  const search = query.search?.trim();
  const category = query.category?.trim();

  if (search && search.length > MAX_SEARCH_LENGTH) {
    throwBadRequest("Search query is too long.");
  }

  if (query.category !== undefined && !category) {
    throwBadRequest("Category cannot be empty.");
  }

  const status = query.status ?? "all";

  if (!["all", "active", "inactive"].includes(status)) {
    throwBadRequest("Invalid product status filter.");
  }

  const stockStatus = query.stockStatus ?? "all";

  if (
    !["all", "in_stock", "low_stock", "out_of_stock"].includes(
      stockStatus
    )
  ) {
    throwBadRequest("Invalid stock status filter.");
  }

  const sort = query.sort ?? "newest";

  if (
    typeof sort !== "string" ||
    !Object.hasOwn(SORT_OPTIONS, sort)
  ) {
    throwBadRequest("Invalid product sorting option.");
  }

  const page =
    query.page === undefined
      ? DEFAULT_PAGE
      : parseInteger(query.page, "Page", 1, 1000000);

  const limit =
    query.limit === undefined
      ? DEFAULT_LIMIT
      : parseInteger(query.limit, "Limit", 1, MAX_LIMIT);

  return {
    search,
    category,
    status,
    stockStatus,
    sort,
    page,
    limit,
  };
};

const getAdminProducts = async (query = {}) => {
  const {
    search,
    category,
    status,
    stockStatus,
    sort,
    page,
    limit,
  } = normalizeAdminProductQuery(query);

  const filter = {};

  if (status === "active") filter.isActive = true;
  if (status === "inactive") filter.isActive = false;

  if (search) {
    const searchRegex = new RegExp(escapeRegex(search), "i");

    filter.$or = [
      { name: searchRegex },
      { sku: searchRegex },
    ];
  }

  if (category) {
    const categoryDocument = mongoose.isValidObjectId(category)
      ? await Category.findOne({ _id: category, isActive: true })
          .select("_id name")
          .lean()
      : await Category.findOne({
          isActive: true,
          $or: [
            { slug: category },
            {
              name: {
                $regex: `^${escapeRegex(category)}$`,
                $options: "i",
              },
            },
          ],
        })
          .select("_id name")
          .lean();

    const categoryFilter = categoryDocument
      ? {
          $or: [
            { categoryId: categoryDocument._id },
            {
              category: categoryDocument.name,
              categoryId: null,
            },
            {
              category: categoryDocument.name,
              categoryId: { $exists: false },
            },
          ],
        }
      : { category };

    filter.$and = [
      ...(filter.$and || []),
      categoryFilter,
    ];
  }

  if (stockStatus === "in_stock") {
    filter.stock = { $gt: LOW_STOCK_THRESHOLD };
  }

  if (stockStatus === "low_stock") {
    filter.stock = { $gt: 0, $lte: LOW_STOCK_THRESHOLD };
  }

  if (stockStatus === "out_of_stock") {
    filter.stock = 0;
  }

  const skip = (page - 1) * limit;

  const [
    products,
    total,
    categoryDocuments,
    legacyCategories,
  ] = await Promise.all([
    Product.find(filter)
      .sort(SORT_OPTIONS[sort])
      .skip(skip)
      .limit(limit)
      .lean(),

    Product.countDocuments(filter),

    Category.find({ isActive: true })
      .select("name")
      .lean(),

    Product.distinct("category"),
  ]);

  const categoryNames = new Map();

  for (const name of [
    ...categoryDocuments.map((item) => item.name),
    ...legacyCategories,
  ]) {
    if (typeof name !== "string" || !name.trim()) continue;

    const normalizedName = name.trim();
    const key = normalizedName.toLocaleLowerCase();

    if (!categoryNames.has(key)) {
      categoryNames.set(key, normalizedName);
    }
  }

  return {
    products,
    categories: [...categoryNames.values()].sort((a, b) =>
      a.localeCompare(b)
    ),
    pagination: {
      page,
      limit,
      total,
      pages: Math.ceil(total / limit),
      hasNextPage: page * limit < total,
      hasPreviousPage: page > 1,
    },
  };
};

const getAdminProductById = async (productId) => {
  if (!mongoose.isValidObjectId(productId)) {
    throwBadRequest("Invalid product ID.");
  }

  const product = await Product.findById(productId).lean();

  if (!product) {
    throw new ApiError(
      HTTP_STATUS.NOT_FOUND,
      "Product not found."
    );
  }

  return product;
};

const updateProductStatus = async (productId, isActive) => {
  if (!mongoose.isValidObjectId(productId)) {
    throwBadRequest("Invalid product ID.");
  }

  const product = await Product.findOneAndUpdate(
    { _id: productId, isActive: !isActive },
    { $set: { isActive } },
    { new: true, runValidators: true }
  ).lean();

  if (!product) {
    const exists = await Product.exists({ _id: productId });

    if (!exists) {
      throw new ApiError(
        HTTP_STATUS.NOT_FOUND,
        "Product not found."
      );
    }

    throw new ApiError(
      HTTP_STATUS.CONFLICT,
      `Product is already ${isActive ? "active" : "inactive"}.`
    );
  }

  return product;
};

const activateProduct = (productId) =>
  updateProductStatus(productId, true);

const deactivateProduct = (productId) =>
  updateProductStatus(productId, false);

const adminService = {
  getDashboardStats,
  getAdminProducts,
  getAdminProductById,
  activateProduct,
  deactivateProduct,
};

export default adminService;

    );
  }

  const parsed = Number(value);

  if (!Number.isSafeInteger(parsed) || parsed < min || parsed > max) {
    throwBadRequest(
      `${field} must be an integer between ${min} and ${max}.`
    );
  }

  return parsed;
};

const normalizeQuery = (query = {}) => {
  const allowedFields = new Set([
    "search",
    "status",
    "sort",
    "page",
    "limit",
  ]);

  for (const field of Object.keys(query)) {
    if (!allowedFields.has(field)) {
      throwBadRequest(`Unsupported query parameter: ${field}.`);
    }
  }

  if (
    query.search !== undefined &&
    typeof query.search !== "string"
  ) {
    throwBadRequest("Search must be a valid string.");
  }

  const search = query.search?.trim() || "";

  if (search.length > MAX_SEARCH_LENGTH) {
    throwBadRequest("Search query is too long.");
  }

  const status = query.status ?? "all";

  if (!["all", "active", "inactive"].includes(status)) {
    throwBadRequest("Invalid customer status filter.");
  }

  const sort = query.sort ?? "newest";

  if (
    typeof sort !== "string" ||
    !Object.hasOwn(SORT_OPTIONS, sort)
  ) {
    throwBadRequest("Invalid customer sorting option.");
  }

  const page =
    query.page === undefined
      ? DEFAULT_PAGE
      : parseInteger(query.page, "Page", 1, 1000000);

  const limit =
    query.limit === undefined
      ? DEFAULT_LIMIT
      : parseInteger(query.limit, "Limit", 1, MAX_LIMIT);

  return { search, status, sort, page, limit };
};

const getAdminCustomers = async (query = {}) => {
  const { search, status, sort, page, limit } = normalizeQuery(query);

  const filter = { role: "customer" };

  if (status === "active") filter.isActive = true;
  if (status === "inactive") filter.isActive = false;

  if (search) {
    const searchRegex = new RegExp(escapeRegex(search), "i");

    filter.$or = [
      { fullName: searchRegex },
      { email: searchRegex },
      { phone: searchRegex },
    ];
  }

  const skip = (page - 1) * limit;

  const [customers, total] = await Promise.all([
    User.find(filter)
      .select("_id fullName email phone role isActive createdAt updatedAt")
      .sort(SORT_OPTIONS[sort])
      .skip(skip)
      .limit(limit)
      .lean(),

    User.countDocuments(filter),
  ]);

  const customerIds = customers.map((customer) => customer._id);

  const orderStats = customerIds.length
    ? await Order.aggregate([
        {
          $match: {
            user: { $in: customerIds },
          },
        },
        {
          $group: {
            _id: "$user",
            totalOrders: { $sum: 1 },
            deliveredOrders: {
              $sum: {
                $cond: [{ $eq: ["$status", "delivered"] }, 1, 0],
              },
            },
            totalOrderValue: { $sum: "$total" },
            cancelledOrderValue: {
              $sum: {
                $cond: [
                  { $eq: ["$status", "cancelled"] },
                  "$total",
                  0,
                ],
              },
            },
          },
        },
      ])
    : [];

  const statsMap = new Map(
    orderStats.map((item) => [item._id.toString(), item])
  );

  const result = customers.map((customer) => {
    const stats = statsMap.get(customer._id.toString());

    return {
      ...customer,
      orderStats: {
        totalOrders: stats?.totalOrders ?? 0,
        deliveredOrders: stats?.deliveredOrders ?? 0,
        totalOrderValue: stats?.totalOrderValue ?? 0,
        cancelledOrderValue: stats?.cancelledOrderValue ?? 0,
      },
    };
  });

  return {
    customers: result,
    pagination: {
      page,
      limit,
      total,
      pages: Math.ceil(total / limit),
      hasNextPage: page * limit < total,
      hasPreviousPage: page > 1,
    },
  };
};

const getAdminCustomerById = async (customerId) => {
  if (!mongoose.isValidObjectId(customerId)) {
    throwBadRequest("Invalid customer ID.");
  }

  const customer = await User.findOne({
    _id: customerId,
    role: "customer",
  })
    .select("_id fullName email phone role isActive createdAt updatedAt")
    .lean();

  if (!customer) {
    throwNotFound("Customer not found.");
  }

  const [summaryResult, recentOrders] = await Promise.all([
    Order.aggregate([
      {
        $match: {
          user: customer._id,
        },
      },
      {
        $group: {
          _id: null,
          totalOrders: { $sum: 1 },
          deliveredOrders: {
            $sum: {
              $cond: [{ $eq: ["$status", "delivered"] }, 1, 0],
            },
          },
          cancelledOrders: {
            $sum: {
              $cond: [{ $eq: ["$status", "cancelled"] }, 1, 0],
            },
          },
          activeOrders: {
            $sum: {
              $cond: [
                {
                  $in: [
                    "$status",
                    ["placed", "confirmed", "processing", "shipped"],
                  ],
                },
                1,
                0,
              ],
            },
          },
          totalOrderValue: { $sum: "$total" },
          cancelledOrderValue: {
            $sum: {
              $cond: [
                { $eq: ["$status", "cancelled"] },
                "$total",
                0,
              ],
            },
          },
        },
      },
    ]),

    Order.find({ user: customer._id })
      .select(
        "orderNumber status paymentStatus paymentMethod subtotal total items createdAt"
      )
      .sort({ createdAt: -1, _id: -1 })
      .limit(RECENT_ORDERS_LIMIT)
      .lean(),
  ]);

  const summary = summaryResult[0] || {
    totalOrders: 0,
    deliveredOrders: 0,
    cancelledOrders: 0,
    activeOrders: 0,
    totalOrderValue: 0,
    cancelledOrderValue: 0,
  };

  return {
    customer,
    summary,
    recentOrders,
  };
};

const updateCustomerStatus = async (customerId, isActive) => {
  if (!mongoose.isValidObjectId(customerId)) {
    throwBadRequest("Invalid customer ID.");
  }

  if (typeof isActive !== "boolean") {
    throwBadRequest("Customer status must be a boolean.");
  }

  const customer = await User.findOneAndUpdate(
    {
      _id: customerId,
      role: "customer",
      isActive: !isActive,
    },
    {
      $set: { isActive },
    },
    {
      new: true,
      runValidators: true,
    }
  )
    .select("_id fullName email phone role isActive createdAt updatedAt")
    .lean();

  if (customer) return customer;

  const exists = await User.exists({
    _id: customerId,
    role: "customer",
  });

  if (!exists) {
    throwNotFound("Customer not found.");
  }

  throw new ApiError(
    HTTP_STATUS.CONFLICT,
    `Customer is already ${isActive ? "active" : "inactive"}.`
  );
};

const adminCustomerService = {
  getAdminCustomers,
  getAdminCustomerById,
  updateCustomerStatus,
};

export default adminCustomerService;
